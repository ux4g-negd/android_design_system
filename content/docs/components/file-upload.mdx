<h2><MIcon name="code" /> Basic Usage</h2>

import com.ux4g.designsystemv1.components.Ux4gFileUpload

Ux4gFileUpload(
    label = "Upload Income Proof (PDF or JPG)",
    helperText = "Maximum file size: 5 MB",
    allowedExtensions = listOf("pdf", "jpg", "png"),
    onFileSelected = { uri -> /* upload file */ },
    isRequired = true
