## Jetpack Compose Usage

```kotlin title="Body Usage"
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import com.ux4g.designsystemv1.theme.Ux4gTheme

@Composable
fun ApplicationDescription() {
    Text(
        text = "Please ensure all scanned identity documents are uploaded in PDF or JPG format before submitting your online application.",
        style = Ux4gTheme.typography.bM_default,
        color = Ux4gTheme.colorScheme.onBackground
    )
}
```
