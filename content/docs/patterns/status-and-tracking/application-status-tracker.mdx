# Application Status Tracker

A high-visibility status monitoring pattern that empowers citizens to track their public service, certificate, or welfare applications through each milestone checkpoint—from initial submission and document verification to officer scrutiny and final certificate issuance.

The pattern adapts dynamically across all key administrative lifecycle states:
- **Under Review**: Standard in-progress state displaying current officer review and SLA milestone progression.
- **Action Required**: Urgent citizen prompt alerting applicant to upload missing or clearer proof documents.
- **Rejected**: Transparent adverse decision screen providing specific rejection reasons, remediation guidance, and single-tap reapplication.
- **Delayed**: Proactive administrative delay advisory citing verification extension and a revised expected completion date with escalation recourse.
- **Approved**: Final success state celebrating application approval with immediate PDF certificate download and DigiLocker synchronization.

            .background(bg)

        verticalAlignment = Alignment.Top
    ) {
        Icon(
            imageVector = Icons.Filled.Error,
            contentDescription = "Action Required",
            tint = text,
            modifier = Modifier.size(18.dp)
        )
        Spacer(modifier = Modifier.width(8.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = "Action required: upload your income proof document",
                style = TextStyle(fontSize = 12.sp, fontWeight = FontWeight.Medium, color = text, lineHeight = 16.sp)
            )
            Spacer(modifier = Modifier.height(6.dp))
            Text(
                text = "Upload Document",
                style = TextStyle(
                    fontSize = 12.sp,
                    fontWeight = FontWeight.Bold,

                modifier = Modifier.clickable(onClick = onUploadDocument)
            )
        }
    }
}

            variant = Ux4gAppHeaderVariant.Standard,
            navigationIcon = Icons.Filled.Menu,
            onNavigationClick = {},
            actions = listOf(
                Ux4gAppHeaderAction(icon = Icons.Filled.Menu, contentDescription = "Menu", onClick = {})
            )
        )

        // Scrollable Card Container
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(scrollState)
                .padding(horizontal = 16.dp, vertical = 12.dp)
        ) {
            // Back Button
            Row(
                modifier = Modifier
                    .clickable(onClick = onBackClick)
                    .padding(vertical = 6.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                    contentDescription = "Back",
                    tint = Ux4gColors.Primary600,
                    modifier = Modifier.size(18.dp)
                )
                Spacer(modifier = Modifier.width(6.dp))
                Text(
                    text = "Back",
                    style = TextStyle(fontSize = 14.sp, fontWeight = FontWeight.SemiBold, color = Ux4gColors.Primary600)
                )
            }

            Spacer(modifier = Modifier.height(8.dp))

            // Contextual Alert Banner
            when (status) {
                ApplicationStatus.ActionRequired -> {
                    ActionRequiredBanner(onUploadDocument = onUploadDocument, isDark = isDark)
                    Spacer(modifier = Modifier.height(12.dp))
                }
                ApplicationStatus.Rejected -> {
                    RejectedBanner(isDark = isDark)
                    Spacer(modifier = Modifier.height(12.dp))
                }
                ApplicationStatus.Delayed -> {
                    DelayedBanner(isDark = isDark)
                    Spacer(modifier = Modifier.height(12.dp))
                }
                else -> Unit
            }

            // Elevated Info Card Container
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .shadow(elevation = 2.dp, shape = RoundedCornerShape(14.dp))
                    .clip(RoundedCornerShape(14.dp))
                    .background(cardBg)
                    .border(1.dp, cardBorder, RoundedCornerShape(14.dp))
                    .padding(16.dp)
            ) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = serviceName,
                        style = TextStyle(fontSize = 16.sp, fontWeight = FontWeight.Bold, color = textPrimary)
                    )
                    Ux4gTag(
                        text = status.title,
                        colorScheme = tagColor,
                        style = Ux4gTagStyle.Tonal
                    )
                }

                Spacer(modifier = Modifier.height(4.dp))

                Text(
                    text = "Application ID · $applicationId",
                    style = TextStyle(fontSize = 12.sp, color = textSubtle)
                )

                if (status != ApplicationStatus.Approved) {
                    Spacer(modifier = Modifier.height(12.dp))
                    Text(
                        text = "8 days left",
                        style = TextStyle(fontSize = 11.sp, fontWeight = FontWeight.Medium, color = textSubtle)
                    )
                    Spacer(modifier = Modifier.height(4.dp))
                    LinearProgressIndicator(
                        progress = { 0.6f },
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(6.dp)
                            .clip(RoundedCornerShape(3.dp)),
                        color = Ux4gColors.Primary600,
                        trackColor = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)
                    )
                }
            }

            Spacer(modifier = Modifier.height(14.dp))

            // Elevated Journey Timeline Card Container
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .shadow(elevation = 2.dp, shape = RoundedCornerShape(14.dp))
                    .clip(RoundedCornerShape(14.dp))
                    .background(cardBg)
                    .border(1.dp, cardBorder, RoundedCornerShape(14.dp))
                    .padding(16.dp)
            ) {
                Text(
                    text = "Application journey",
                    style = TextStyle(fontSize = 15.sp, fontWeight = FontWeight.Bold, color = textPrimary)
                )
                Text(
                    text = "Every step from submission to issuance",
                    style = TextStyle(fontSize = 12.sp, color = textSubtle)
                )

                Spacer(modifier = Modifier.height(16.dp))

                val isApproved = status == ApplicationStatus.Approved

                val steps = listOf(
                    Ux4gJourneyStep(
                        state = Ux4gJourneyStepState.Completed,
                        date = "07 Apr 2026, 10:24 AM",
                        title = "Application Submitted"
                    ),
                    Ux4gJourneyStep(
                        state = Ux4gJourneyStepState.Completed,
                        date = "10 Apr 2026, 02:15 PM",
                        title = "Documents Verified"
                    ),
                    Ux4gJourneyStep(
                        state = if (isApproved) Ux4gJourneyStepState.Completed else Ux4gJourneyStepState.Current,
                        date = "11 Apr 2026, 09:03 AM",
                        title = "Officer Review"
                    ),
                    Ux4gJourneyStep(
                        state = if (isApproved) Ux4gJourneyStepState.Current else Ux4gJourneyStepState.Upcoming,
                        date = if (isApproved) "14 Apr 2026, 11:30 AM" else "Est. 22 Apr 2026",
                        title = if (isApproved) "Certificate Issued" else "Certificate Will Be Issued",
                        status = if (!isApproved) Ux4gJourneyStepStatus(
                            text = if (status == ApplicationStatus.UnderReview) "2 days remaining" else "11 days remaining",
                            badgeText = "Pending",
                            badgePosition = Ux4gJourneyBadgePosition.Bottom
                        ) else null
                    )
                )

                Ux4gJourneyTimeline(
                    steps = steps,
                    orientation = Ux4gJourneyOrientation.Vertical
                )
            }

            Spacer(modifier = Modifier.height(18.dp))

            // Contextual Actions
            when (status) {
                ApplicationStatus.Rejected -> {
                    Ux4gButton(
                        text = "Reapply",
                        variant = Ux4gButtonVariant.Primary,
                        size = Ux4gButtonSize.Large,
                        onClick = onReapply,
                        modifier = Modifier.fillMaxWidth()
                    )
                    Spacer(modifier = Modifier.height(10.dp))
                    TextButton(
                        onClick = onContactOffice,
                        modifier = Modifier.align(Alignment.CenterHorizontally)
                    ) {
                        Text(
                            text = "Contact district office",
                            style = TextStyle(fontSize = 14.sp, color = Ux4gColors.Primary600, fontWeight = FontWeight.SemiBold)
                        )
                    }
                }
                ApplicationStatus.Delayed -> {
                    TextButton(
                        onClick = onEscalate,
                        modifier = Modifier.align(Alignment.CenterHorizontally)
                    ) {
                        Text(
                            text = "Escalate this application",
                            style = TextStyle(fontSize = 14.sp, color = Color(0xFFDC2626), fontWeight = FontWeight.SemiBold)
                        )
                    }
                }
                ApplicationStatus.Approved -> {
                    Ux4gButton(
                        text = "Download Certificate (PDF)",
                        variant = Ux4gButtonVariant.Primary,
                        size = Ux4gButtonSize.Large,
                        onClick = onDownloadPdf,
                        modifier = Modifier.fillMaxWidth()
                    )
                    Spacer(modifier = Modifier.height(10.dp))
                    Ux4gButton(
                        text = "Save to DigiLocker",
                        variant = Ux4gButtonVariant.Outline,
                        size = Ux4gButtonSize.Large,
                        onClick = onSaveDigiLocker,
                        modifier = Modifier.fillMaxWidth()
                    )
                }
                else -> Unit
            }

            Spacer(modifier = Modifier.height(16.dp))
        }

        // Footer
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(vertical = 12.dp),
            horizontalArrangement = Arrangement.Center,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = "Powered by Digital India",
                style = TextStyle(
                    fontSize = 11.sp,
                    color = Ux4gColors.Neutral400,
                    fontWeight = FontWeight.Medium
                )
            )
        }
    }
