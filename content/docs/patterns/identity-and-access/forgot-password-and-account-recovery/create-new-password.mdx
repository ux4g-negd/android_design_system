# Create new password

A secure credential update pattern for citizen account recovery. The screen features a branded government app header, prompt instruction (*"Choose a strong password to secure your account"*), a Password input with visibility toggle, a live password strength indicator bar with label (*"Password Strength: Strong"*), an interactive requirement checklist (8+ characters, Uppercase letter, Number, Special character), a Confirm password input with live mismatch error validation (*"Passwords do not match"*), a primary **Reset password** action button, and a pinned Digital India brand footer.

@Composable
private fun RequirementItem(
    text: String,
    isSatisfied: Boolean,
    colors: CreateNewPasswordColors
) {
    Row(verticalAlignment = Alignment.CenterVertically) {
        if (isSatisfied) {
            Box(
                modifier = Modifier
                    .size(18.dp)
                    .clip(CircleShape)
                    .background(colors.validCheckIcon),
                contentAlignment = Alignment.Center
            ) {
                Icon(
                    imageVector = Icons.Filled.Check,
                    contentDescription = null,
                    tint = Color.White,
                    modifier = Modifier.size(12.dp)
                )
            }
        } else {
            Box(
                modifier = Modifier
                    .size(18.dp)
                    .clip(CircleShape)
                    .background(colors.invalidErrorIcon),
                contentAlignment = Alignment.Center
            ) {
                Text(
                    text = "!",
                    style = TextStyle(
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold,
                        color = Color.White
                    )
                )
            }
        }
        Spacer(Modifier.width(10.dp))
        Text(
            text = text,
            style = TextStyle(
                fontSize = 14.sp,
                color = colors.subtleText
            )
        )
    }
}
