# Account recovery

An account recovery pattern enabling citizens to recover account access and initiate credential renewal using their Aadhaar Number when mobile numbers or passwords are unknown. The screen features a **Back to Sign in** navigation link, prominent heading (*"Account recovery"*), an auto-formatting Aadhaar Number input with digit masking toggle, an explicit verification consent checkbox (*"I agree to verify my identity via Aadhaar OTP for the purpose of password recovery."*), a primary **Send OTP** call-to-action button, and a pinned Digital India brand footer.

@Composable
fun AccountRecoveryScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    onMenuClick: () -> Unit = {},
    onBackToSignIn: () -> Unit = {},
    onSendOtp: (String) -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkAccountRecoveryColors else LightAccountRecoveryColors

    var aadhaarNumber by rememberSaveable { mutableStateOf("XXXX XXXX 1234") }
    var isConsentChecked by rememberSaveable { mutableStateOf(false) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 24.dp, vertical = 24.dp)
        ) {
            // Back to Sign in link
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier
                    .clip(RoundedCornerShape(6.dp))
                    .clickable(onClick = onBackToSignIn)
                    .padding(vertical = 4.dp)
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                    contentDescription = "Back to Sign in",
                    tint = colors.backAction,
                    modifier = Modifier.size(18.dp)
                )
                Spacer(Modifier.width(6.dp))
                Text(
                    text = "Back to Sign in",
                    style = TextStyle(
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium,
                        color = colors.backAction
                    )
                )
            }
            Spacer(Modifier.height(28.dp))

            // Heading
            Text(
                text = "Account recovery",
                style = TextStyle(
                    fontSize = 30.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.title,
                    letterSpacing = (-0.4).sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // Aadhaar Number Input with Mask Toggle
            Ux4gAadhaarInputField(
                value = aadhaarNumber,
                onValueChange = { aadhaarNumber = it },
                label = "Aadhaar Number",
                placeholder = "XXXX XXXX 1234",
                showMaskToggle = true,
                size = Ux4gInputFieldSize.Large,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(20.dp))

            // Identity Verification Consent Checkbox
            Ux4gCheckbox(
                value = isConsentChecked,
                onValueChange = { isConsentChecked = it == true },
                label = "I agree to verify my identity via Aadhaar OTP for the purpose of password recovery.",
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(28.dp))

            // Primary Call to Action Button
            Ux4gButton(
                text = "Send OTP",
                onClick = { onSendOtp(aadhaarNumber) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(24.dp))
        }

        BrandFooter(colors = colors)
    }
}

@Composable
private fun BrandHeader(colors: AccountRecoveryColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16,
            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}

            contentDescription = "Digital India",
            colorFilter = colors.footerLogoTint?.let { ColorFilter.tint(it) },

        )
    }
}
