# Enter OTP

An account recovery OTP verification pattern for citizen credential reset. The screen features a branded government app header, a top **"Change mobile number"** navigation link, destination display (*"Sent to +91 98765 XXXXX"*), 6 auto-advancing `Ux4gOtpInput` digit boxes with integrated reverse countdown timer (*"Didn't receive OTP? Resend in 00:17"*), primary **Verify OTP** action button, and a pinned Digital India brand footer.

@Composable
fun RecoveryEnterOtpScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    destinationText: String = "Sent to +91 98765 XXXXX",
    onChangeMobileNumber: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyOtp: (otp: String) -> Unit = {},
    onResendOtp: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkRecoveryEnterOtpColors else LightRecoveryEnterOtpColors
    val typography = Ux4gTheme.typography

    var otp by rememberSaveable { mutableStateOf("") }
    var resendNonce by remember { mutableIntStateOf(0) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = Ux4gSpace.Space20, end = Ux4gSpace.Space20, top = Ux4gSpace.Space20)
        ) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier
                    .clip(RoundedCornerShape(8.dp))
                    .clickable(onClick = onChangeMobileNumber)
                    .padding(vertical = 4.dp, horizontal = 2.dp)
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                    contentDescription = "Change mobile number",
                    tint = Ux4gColors.primary600,
                    modifier = Modifier.size(20.dp)
                )
                Spacer(Modifier.width(6.dp))
                Text(
                    text = "Change mobile number",
                    style = TextStyle(
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium,
                        color = Ux4gColors.primary600
                    )
                )
            }
            Spacer(Modifier.height(24.dp))

            Text(
                text = "Enter OTP",
                style = typography.hM_strong.copy(
                    fontSize = 28.sp,
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))
            Text(
                text = destinationText,
                style = typography.bM_default.copy(color = colors.subtleText)
            )
            Spacer(Modifier.height(Ux4gSpace.Space24))

            // 6-digit auto-advancing OTP digit boxes with countdown timer
            Ux4gOtpInput(
                value = otp,
                onValueChange = { otp = it },
                length = 6,
                captionVariant = Ux4gOtpCaptionVariant.ResendTimer,
                captionLeadingText = "Didn't receive OTP?",
                captionTrailingText = "Resend in 00:17",
                autoCountdownSeconds = 60,
                onCaptionTrailingTap = {
                    otp = ""
                    resendNonce++
                    onResendOtp()
                },
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space28))

            Ux4gButton(
                text = "Verify OTP",
                onClick = { onVerifyOtp(otp) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(Ux4gSpace.Space20))
        }

        BrandFooter(colors = colors)
    }
}
