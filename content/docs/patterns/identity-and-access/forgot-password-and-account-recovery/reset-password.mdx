# Reset Password

An essential identity recovery pattern for citizens who have forgotten their credentials. The screen features a branded government app header, a top **"Back to Sign in"** navigation link, prompt instructions (*"Enter your registered mobile number to receive a verification code"*), a Mobile Number input with **+91** country prefix, a primary **Send OTP** action button, an inline **"Recover account using Aadhaar Number"** fallback with a trailing arrow, an **OR** divider, alternative sign-in options, and a pinned Digital India brand footer.

@Composable
fun ResetPasswordScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialMobileNumber: String = "",
    onBackToSignIn: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onSendOtp: (mobileNumber: String) -> Unit = {},
    onRecoverWithAadhaar: () -> Unit = {},
    onSignInWithEmail: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkResetPasswordColors else LightResetPasswordColors
    val typography = Ux4gTheme.typography

    var mobileNumber by rememberSaveable { mutableStateOf(initialMobileNumber) }

    val placeholderStyle = TextStyle(
        fontSize = 14.sp,
        fontWeight = FontWeight.Normal,
        color = colors.mutedText,
        lineHeight = 18.sp
    )

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = Ux4gSpace.Space20, end = Ux4gSpace.Space20, top = Ux4gSpace.Space20)
        ) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier
                    .clip(RoundedCornerShape(8.dp))
                    .clickable(onClick = onBackToSignIn)
                    .padding(vertical = 4.dp, horizontal = 2.dp)
            ) {
                Icon(
                    imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                    contentDescription = "Back to Sign in",
                    tint = Ux4gColors.primary600,
                    modifier = Modifier.size(20.dp)
                )
                Spacer(Modifier.width(6.dp))
                Text(
                    text = "Back to Sign in",
                    style = TextStyle(
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium,
                        color = Ux4gColors.primary600
                    )
                )
            }
            Spacer(Modifier.height(24.dp))

            Text(
                text = "Reset Password",
                style = typography.hM_strong.copy(
                    fontSize = 28.sp,
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))
            Text(
                text = "Enter your registered mobile number to receive a verification code",
                style = typography.bM_default.copy(color = colors.subtleText)
            )
            Spacer(Modifier.height(Ux4gSpace.Space24))

            Ux4gInputField(
                value = mobileNumber,
                onValueChange = { if (it.length <= 10) mobileNumber = it },
                label = "Mobile Number",
                placeholder = "Enter mobile number",
                placeholderStyle = placeholderStyle,
                type = Ux4gInputFieldType.Number,
                prefixText = "+91",
                maxLength = 10,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space24))

            Ux4gButton(
                text = "Send OTP",
                onClick = { onSendOtp(mobileNumber) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Row(
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = Arrangement.Center,
                modifier = Modifier
                    .fillMaxWidth()
                    .clip(RoundedCornerShape(8.dp))
                    .clickable(onClick = onRecoverWithAadhaar)
                    .padding(vertical = 8.dp)

@Composable
private fun BrandHeader(colors: ResetPasswordColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16,
            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}
