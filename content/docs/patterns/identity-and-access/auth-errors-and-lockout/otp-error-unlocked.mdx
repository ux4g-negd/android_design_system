# OTP error - unlocked

A recovery state pattern displayed when a citizen's temporary account lockout duration has concluded. The screen restores standard authentication access, featuring a prominent green confirmation status banner (*"Your account is unlocked now."*), fresh `Ux4gOtpInput` digit boxes (`Ux4gOtpInputStatus.Default`), an inline resend countdown timer caption (*"Didn't receive OTP? Resend OTP in 00:17"*), a full-width **Verify OTP** button, and a pinned Digital India brand footer.

@Composable
fun AuthOtpUnlockedScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialOtp: String = "",
    maskedPhone: String = "+91 98XXX XXXXX",
    resendTimerText: String = "Resend OTP in 00:17",
    unlockedMessage: String = "Your account is unlocked now.",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyClick: (String) -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkAuthOtpUnlockedColors else LightAuthOtpUnlockedColors
    var otpValue by remember { mutableStateOf(initialOtp) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 24.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(16.dp))

            // Inline green confirmation banner indicating unlocked status
            Ux4gStatusBanner(
                variant = Ux4gBannerVariant.SuccessLight,
                title = unlockedMessage,
                titleStyle = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Medium,
                    color = colors.successText,
                    lineHeight = 18.sp
                ),
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Filled.CheckCircle,
                        contentDescription = null,
                        tint = colors.successIcon,
                        modifier = Modifier.size(20.dp)
                    )
                },
                backgroundColor = colors.successBackground,
                borderColor = colors.successBorder,
                margin = PaddingValues(0.dp),
                padding = PaddingValues(12.dp)
            )
            Spacer(Modifier.height(24.dp))

            Text(
                text = "OTP Verification",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 28.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            Text(
                text = "Enter the 6-digit code sent to $maskedPhone",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit re-enabled OTP input in default active state
            Ux4gOtpInput(
                value = otpValue,
                onValueChange = { otpValue = it },
                length = 6,
                status = Ux4gOtpInputStatus.Default,
                captionVariant = Ux4gOtpCaptionVariant.AttemptWithTimer,
                captionLeadingText = "Didn't receive OTP?",
                captionTrailingText = resendTimerText,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(24.dp))
        }

        // Bottom pinned verify button
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 20.dp, vertical = 12.dp)
        ) {
            Ux4gButton(
                text = "Verify OTP",
                onClick = { onVerifyClick(otpValue) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
        }

        BrandFooter(colors = colors)
    }
}
