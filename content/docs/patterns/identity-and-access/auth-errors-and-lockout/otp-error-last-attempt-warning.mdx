# OTP error - last-attempt warning

A high-urgency authentication error pattern displayed when a citizen inputs an incorrect one-time password (OTP) and reaches the **final attempt** before account lockout. The screen features red-bordered `Ux4gOtpInput` digit boxes (`Ux4gOtpInputStatus.Error`), an inline retry timer caption, an escalated red `Ux4gStatusBanner` (`Ux4gBannerVariant.ErrorLight`) with remaining attempt badge (*"Attempt 2 of 3"*) and high-priority lockout advisory (*"This is your last attempt before a 30-min lockout"*), full-width **Verify OTP** button, and a pinned Digital India brand footer.

@Composable

    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialOtp: String = "555555",
    maskedPhone: String = "+91 98XXX XXXXX",
    resendTimerText: String = "Resend OTP in 00:17",
    attemptText: String = "Attempt 2 of 3",
    warningDescription: String = "This is your last attempt\\nbefore a 30-min lockout",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyClick: (String) -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkAuthOtpLastAttemptColors else LightAuthOtpLastAttemptColors
    var otpValue by remember { mutableStateOf(initialOtp) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 24.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(24.dp))

            Text(
                text = "OTP Verification",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 28.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            Text(
                text = "Enter the 6-digit code sent to $maskedPhone",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit OTP input in error state with red border and timer caption
            Ux4gOtpInput(
                value = otpValue,
                onValueChange = { otpValue = it },
                length = 6,
                status = Ux4gOtpInputStatus.Error,
                captionVariant = Ux4gOtpCaptionVariant.AttemptWithTimer,
                captionLeadingText = "Incorrect OTP",
                captionTrailingText = resendTimerText,

            Spacer(Modifier.height(16.dp))

                title = "Incorrect OTP",

                    fontWeight = FontWeight.Normal,
                    color = colors.errorText,
                    lineHeight = 18.sp
                ),
                leadingIcon = {

                        imageVector = Icons.Filled.Error,
