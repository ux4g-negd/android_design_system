# OTP step-up -- suspicious activity

A security step-up verification challenge displayed when anomalous behavior, unfamiliar location, or new device sign-in is detected during authentication. The screen presents an amber warning status banner (`Ux4gBannerVariant.WarningLight`) detailing the security trigger (*"Suspicious activity detected"* and *"We detected a sign-in from a new device. Please verify your identity before continuing."*), clean 6-digit `Ux4gOtpInput` digit boxes, inline countdown resend caption (*"Didn't receive OTP? Resend in 00:17"*), full-width primary **Verify OTP** button, and a pinned Digital India brand footer.

@Composable
fun AuthOtpSuspiciousActivityScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialOtp: String = "",
    warningTitle: String = "Suspicious activity detected",
    warningDescription: String = "We detected a sign-in from a new device. Please verify your identity before continuing.",
    subtitleText: String = "A verification code has been sent to your registered number for security verification.",
    resendTimerText: String = "Resend in 00:17",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyClick: (String) -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkAuthOtpSuspiciousActivityColors else LightAuthOtpSuspiciousActivityColors
    var otpValue by remember { mutableStateOf(initialOtp) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 24.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(16.dp))

            // Inline warning status banner explaining step-up security rationale
            Ux4gStatusBanner(
                variant = Ux4gBannerVariant.WarningLight,
                title = warningTitle,
                titleStyle = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.warningText,
                    lineHeight = 18.sp
                ),
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Filled.Warning,
                        contentDescription = null,
                        tint = colors.warningIcon,
                        modifier = Modifier.size(20.dp)
                    )
                },
                subtitleContent = {
                    Text(
                        text = warningDescription,
                        style = TextStyle(
                            fontSize = 13.sp,
                            fontWeight = FontWeight.Normal,
                            color = colors.warningText,
                            lineHeight = 17.sp
                        ),
                        modifier = Modifier.fillMaxWidth().padding(top = 4.dp)
                    )
                },
                backgroundColor = colors.warningBackground,
                borderColor = colors.warningBorder,
                margin = PaddingValues(0.dp),
                padding = PaddingValues(12.dp)
            )
            Spacer(Modifier.height(24.dp))

            Text(
                text = "OTP Verification",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 28.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            Text(
                text = subtitleText,
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit OTP input in default active state
            Ux4gOtpInput(
                value = otpValue,
                onValueChange = { otpValue = it },
                length = 6,
                status = Ux4gOtpInputStatus.Default,
                captionVariant = Ux4gOtpCaptionVariant.AttemptWithTimer,
                captionLeadingText = "Didn't receive OTP?",
                captionTrailingText = resendTimerText,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(24.dp))
        }

        // Bottom pinned verify button
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 20.dp, vertical = 12.dp)
        ) {
            Ux4gButton(
                text = "Verify OTP",
                onClick = { onVerifyClick(otpValue) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
        }

        BrandFooter(colors = colors)
    }
}

        )
    }
}
