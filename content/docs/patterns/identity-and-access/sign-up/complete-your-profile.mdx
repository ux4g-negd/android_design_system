# Complete your profile

An onboarding pattern for capturing citizen profile details during the sign-up flow. The screen features a branded government app header, Full name text input, Email Address field, pre-filled/verified Mobile Number with **+91** country prefix, Category single-selection dropdown (`Ux4gSelectionDropdown`), primary **Continue** action button, and a pinned Digital India brand footer.

@Composable
fun CompleteYourProfileScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialFullName: String = "",
    initialEmail: String = "",
    initialMobileNumber: String = "",
    categoryOptions: List<Ux4gDropdownOption> = defaultCategories,
    onMenuClick: () -> Unit = {},
    onContinue: (fullName: String, email: String, mobile: String, categoryId: String?) -> Unit = { _, _, _, _ -> }
) {
    val colors = if (isDarkTheme) DarkCompleteProfileColors else LightCompleteProfileColors
    val typography = Ux4gTheme.typography

    var fullName by rememberSaveable { mutableStateOf(initialFullName) }
    var email by rememberSaveable { mutableStateOf(initialEmail) }
    var mobileNumber by rememberSaveable { mutableStateOf(initialMobileNumber) }
    var selectedCategoryIds by rememberSaveable { mutableStateOf<List<String>>(emptyList()) }

    val placeholderStyle = TextStyle(
        fontSize = 14.sp,
        fontWeight = FontWeight.Normal,
        color = colors.mutedText,
        lineHeight = 18.sp
    )

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = Ux4gSpace.Space20, end = Ux4gSpace.Space20, top = Ux4gSpace.Space24)
        ) {
            Text(
                text = "Complete your profile",
                style = typography.hM_strong.copy(
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(Ux4gSpace.Space24))

            Ux4gInputField(
                value = fullName,
                onValueChange = { fullName = it },
                label = "Full name",
                placeholder = "Enter your full name",
                placeholderStyle = placeholderStyle,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Ux4gInputField(
                value = email,
                onValueChange = { email = it },
                label = "Email Address",
                placeholder = "example@mail.com",
                placeholderStyle = placeholderStyle,
                type = Ux4gInputFieldType.Email,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Ux4gInputField(
                value = mobileNumber,
                onValueChange = { if (it.length <= 10) mobileNumber = it },
                label = "Mobile Number",
                placeholder = "Enter mobile number",
                placeholderStyle = placeholderStyle,
                type = Ux4gInputFieldType.Number,
                prefixText = "+91",
                maxLength = 10,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Ux4gSelectionDropdown(
                label = "Category",
                placeholder = "Please select..",
                options = categoryOptions,
                selectedOptionIds = selectedCategoryIds,
                onSelectionChange = { selectedCategoryIds = it },
                mode = Ux4gDropdownMode.Single,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space28))

            Ux4gButton(
                text = "Continue",
                onClick = {
                    onContinue(fullName, email, mobileNumber, selectedCategoryIds.firstOrNull())
                },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(Ux4gSpace.Space20))
        }

        BrandFooter(colors = colors)
    }
}

@Composable
private fun BrandHeader(colors: CompleteProfileColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16,
            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}

@Composable
private fun BrandFooter(colors: CompleteProfileColors) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(top = Ux4gSpace.Space8, bottom = Ux4gSpace.Space20),
        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = "Powered by -",
            style = Ux4gTheme.typography.lS_default.copy(color = colors.mutedText)
        )
        Spacer(Modifier.width(Ux4gSpace.Space8))
        Image(
            painter = painterResource(AppR.drawable.ic_digital_india),
            contentDescription = "Digital India",
            colorFilter = colors.footerLogoTint?.let { ColorFilter.tint(it) },
            modifier = Modifier.height(22.dp)
        )
    }
}
