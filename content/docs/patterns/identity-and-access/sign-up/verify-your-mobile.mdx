# Verify your mobile

A focused mobile OTP verification pattern for citizen account sign-up. The screen features a branded government header, two-line heading typography (*"Verify your mobile\nnumber"*), masked mobile phone display (*"OTP sent to +91 98765 XXXXX"*), 6 auto-advancing `Ux4gOtpInput` digit boxes with an integrated reverse countdown timer (*"Didn't receive OTP? Resend in 00:17"*), primary **Verify OTP** action, **Change mobile number** link, and a pinned Digital India brand footer.

@Composable
fun VerifyYourMobileScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    destinationText: String = "OTP sent to +91 98765 XXXXX",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyOtp: (otp: String) -> Unit = {},
    onResendOtp: () -> Unit = {},
    onChangeMobileNumber: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkVerifyMobileColors else LightVerifyMobileColors

    var otp by rememberSaveable { mutableStateOf("") }
    var resendNonce by remember { mutableIntStateOf(0) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

                    .clickable(onClick = onBackClick)

                    contentDescription = "Back",
                    tint = Ux4gColors.primary600,

                    text = "Back",
                    style = TextStyle(
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium,
                        color = Ux4gColors.primary600
                    )
                )
            }
            Spacer(Modifier.height(24.dp))

            Text(
                text = "Verify your mobile\nnumber",
                style = TextStyle(
                    fontSize = 26.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 32.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))
            Text(
                text = destinationText,
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit auto-advancing OTP digit boxes with countdown timer
            Ux4gOtpInput(
                value = otp,
                onValueChange = { otp = it },
                length = 6,
                captionVariant = Ux4gOtpCaptionVariant.ResendTimer,

                captionTrailingText = "Resend in 00:17",
                autoCountdownSeconds = 60,
                onCaptionTrailingTap = {

                    resendNonce++
                    onResendOtp()
                },
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(24.dp))

                text = "Verify OTP",

                size = Ux4gButtonSize.Large,

                    size = Ux4gButtonSize.Small,

                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(

                            Icon(
                                imageVector = Icons.Filled.Menu,
