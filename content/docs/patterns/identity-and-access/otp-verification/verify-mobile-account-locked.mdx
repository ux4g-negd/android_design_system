# Verify mobile - account locked

A secure terminal lockout state displayed after a citizen exhausts all available OTP verification attempts. The screen features a prominent red lock badge, a centered advisory header (*"Too many incorrect\nattempts"*), 6 disabled OTP digit boxes with a live lockout timer (*"Locked for 28:43"*), an escalated error banner, a support phone hotline CTA (*"Call 1800-XXX-XXXX"*), and a pinned Digital India brand footer.

@Composable
fun VerifyMobileAccountLockedScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    lockoutDurationText: String = "Locked for 28:43",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onCallSupport: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkVerifyAccountLockedColors else LightVerifyAccountLockedColors

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 24.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(24.dp))

            // Centered Lock badge icon
            Box(
                modifier = Modifier
                    .fillMaxWidth(),
                contentAlignment = Alignment.Center
            ) {
                Box(
                    modifier = Modifier
                        .size(64.dp)
                        .clip(CircleShape)
                        .background(colors.lockBadgeBackground),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        imageVector = Icons.Filled.Lock,
                        contentDescription = "Account Locked",
                        tint = colors.lockBadgeIcon,
                        modifier = Modifier.size(28.dp)
                    )
                }
            }
            Spacer(Modifier.height(16.dp))

            Text(
                text = "Too many incorrect\nattempts",
                style = TextStyle(
                    fontSize = 22.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 26.sp,
                    letterSpacing = (-0.3).sp
                ),
                textAlign = TextAlign.Center,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit disabled OTP input boxes with locked caption state
            Ux4gOtpInput(
                value = "",
                onValueChange = {},
                enabled = false,
                length = 6,
                captionVariant = Ux4gOtpCaptionVariant.Locked,
                captionLeadingText = lockoutDurationText,
                captionTrailingText = "Resend OTP",
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(12.dp))

            // Error banner indicating lockout duration
            Ux4gStatusBanner(
                variant = Ux4gBannerVariant.ErrorLight,
                title = "Account locked. Please wait for the\ncountdown to complete",
                titleStyle = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Normal,
                    color = colors.errorText,
                    lineHeight = 18.sp
                ),
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Filled.Error,
                        contentDescription = null,
                        tint = colors.errorIcon,
                        modifier = Modifier.size(20.dp)
                    )
                },
                backgroundColor = colors.errorBackground,
                borderColor = colors.errorBorder,
                margin = PaddingValues(0.dp),
                padding = PaddingValues(12.dp)
            )
            Spacer(Modifier.height(20.dp))

            // Support hotline row
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "Need help? ",
                    style = TextStyle(

        BrandFooter(colors = colors)
    }
}
