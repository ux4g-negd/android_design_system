# Verify mobile with voice fallback

A resilient mobile OTP verification pattern for citizen authentication when SMS delivery is delayed or unreliable. The screen features a branded government header, a ghost-style **Back Button**, multi-line title typography (*"Verify your mobile\nnumber"*), masked mobile destination, 6-digit `Ux4gOtpInput` with active tap-able **Resend OTP** action, primary **Verify OTP** button, an accessible secondary **Get OTP via voice call** outline action, and a pinned Digital India brand footer.

@Composable
fun VerifyMobileWithVoiceFallbackScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    destinationText: String = "OTP sent to +91 98765 XXXXX",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyOtp: (otp: String) -> Unit = {},
    onResendOtp: () -> Unit = {},
    onVoiceCallOtp: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkVerifyVoiceFallbackColors else LightVerifyVoiceFallbackColors

    var otp by rememberSaveable { mutableStateOf("") }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 32.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(24.dp))

            Text(
                text = "Verify your mobile\nnumber",
                style = TextStyle(
                    fontSize = 26.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 32.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))
            Text(
                text = destinationText,
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit OTP input with immediate tap-able Resend Action (timer expired)
            Ux4gOtpInput(
                value = otp,
                onValueChange = { otp = it },
                length = 6,
                captionVariant = Ux4gOtpCaptionVariant.ResendAction,
                captionLeadingText = "Didn't receive OTP?",
                captionTrailingText = "Resend OTP",
                onCaptionTrailingTap = {
                    otp = ""
                    onResendOtp()
                },
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(24.dp))

            Ux4gButton(
                text = "Verify OTP",
                onClick = { onVerifyOtp(otp) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(12.dp))

            // Secondary voice fallback action button
            Ux4gButton(
                text = "Get OTP via voice call",
                onClick = onVoiceCallOtp,
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Large,
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Outlined.Phone,
                        contentDescription = null,
                        modifier = Modifier.size(18.dp)
                    )
                },
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
        }

        BrandFooter(colors = colors)
    }
}

@Composable

        modifier = Modifier.height(48.dp)
    )
}

                        height = 28.dp

                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)

                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)

        Ux4gDivider(color = colors.border)

        )
