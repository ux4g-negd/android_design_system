# Verify mobile with last-attempt warning

A high-urgency mobile OTP verification pattern for citizen authentication on the **final attempt**. The screen features an escalated red error status banner, explicit lockout consequence messaging (*"This is your last attempt before a 30-min lockout"*), an updated attempt meter (*"Attempt 2 of 3"*), a primary **Verify OTP** action, and a pinned Digital India brand footer.

@Composable
fun VerifyMobileWithLastAttemptWarningScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    destinationText: String = "OTP sent to +91 98765 XXXXX",
    attempt: Int = 2,
    maxAttempts: Int = 3,
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onVerifyOtp: (otp: String) -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkVerifyLastAttemptColors else LightVerifyLastAttemptColors

    var otp by rememberSaveable { mutableStateOf("") }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 32.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(24.dp))

            Text(
                text = "Verify your mobile\nnumber",
                style = TextStyle(
                    fontSize = 26.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 32.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))
            Text(
                text = destinationText,
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // 6-digit OTP input boxes
            Ux4gOtpInput(
                value = otp,
                onValueChange = { otp = it },
                length = 6,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(16.dp))

            // Escalated Error banner for the final attempt before lockout
            Ux4gStatusBanner(
                variant = Ux4gBannerVariant.ErrorLight,
                title = "Incorrect OTP",
                titleStyle = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Normal,
                    color = colors.errorText,
                    lineHeight = 18.sp
                ),
                subtitleWidget = {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text(
                            text = "This is your last attempt\nbefore a 30-min lockout",
                            style = TextStyle(
                                fontSize = 14.sp,
                                fontWeight = FontWeight.Bold,
                                color = colors.errorText,
                                lineHeight = 18.sp
                            ),
                            modifier = Modifier.weight(1f)
                        )
                        Text(
                            text = "Attempt $attempt of $maxAttempts",
                            style = TextStyle(
                                fontSize = 12.sp,
                                fontWeight = FontWeight.Medium,
                                color = colors.errorText,
                                lineHeight = 14.sp
                            ),
                            modifier = Modifier
                                .clip(RoundedCornerShape(6.dp))
                                .background(colors.attemptPillBackground)
                                .padding(horizontal = 10.dp, vertical = 5.dp)
                        )
                    }
                },
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Filled.Error,
                        contentDescription = null,
                        tint = colors.errorIcon,
                        modifier = Modifier.size(20.dp)
                    )
                },
                backgroundColor = colors.errorBackground,
                borderColor = colors.errorBorder,
                margin = PaddingValues(0.dp),
                padding = PaddingValues(start = 12.dp, top = 12.dp, end = 10.dp, bottom = 12.dp)
            )
            Spacer(Modifier.height(20.dp))

            Ux4gButton(
                text = "Verify OTP",
                onClick = { onVerifyOtp(otp) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
        }

        BrandFooter(colors = colors)
    }
}

private fun BrandHeader(colors: VerifyLastAttemptColors, onMenuClick: () -> Unit) {

                    customWidget = {

        Ux4gDivider(color = colors.border)
    }
}

        )
    }
}`}
