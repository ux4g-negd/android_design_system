# Signed in success

A clean, reassuring sign-in success state for citizen mobile services. The screen provides instant positive visual feedback through a concentric green checkmark badge, confirmation typography (*"Signed in successfully!"*), service redirection messaging, a 3-dot animated loading indicator, a dynamic countdown timer (*"Redirecting in 3 seconds..."*), and a pinned Digital India brand footer.

@Composable
fun SignedInSuccessScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialSeconds: Int = 3,
    onMenuClick: () -> Unit = {},
    onRedirect: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkSignedInSuccessColors else LightSignedInSuccessColors

    var secondsLeft by remember { mutableIntStateOf(initialSeconds) }

    LaunchedEffect(Unit) {
        while (secondsLeft > 0) {
            delay(1000L)
            secondsLeft--
        }
        onRedirect()
    }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)

            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            // Concentric circle success badge with checkmark icon
            Box(
                modifier = Modifier
                    .size(64.dp)
                    .clip(CircleShape)
                    .background(colors.successOuterCircle),
                contentAlignment = Alignment.Center
            ) {
                Box(

                        tint = Color.White,
                        modifier = Modifier.size(22.dp)
                    )
                }
            }
            Spacer(Modifier.height(24.dp))

                text = "Signed in successfully!",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.successTitle,
                    lineHeight = 28.sp,

            )
            Spacer(Modifier.height(8.dp))

                textAlign = TextAlign.Center
            )
        }

        BrandFooter(colors = colors)
    }
}

@Composable
private fun BrandHeader(colors: SignedInSuccessColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,

            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {

                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}
