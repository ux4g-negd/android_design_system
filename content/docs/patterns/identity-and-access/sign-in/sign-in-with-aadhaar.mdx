# Sign in with Aadhaar

A standardized Aadhaar sign-in pattern for citizen identity access. The screen combines a branded government header, a back navigation button, formatted 12-digit Aadhaar number input with instant Verhoeff checksum validation, authentication method selection (**Send OTP** vs **Face Auth**), conditional **Continue** button activation, an encryption privacy assurance (*"Your Aadhaar details are encrypted and secure"*), and a pinned Digital India brand footer.

@Composable
fun SignInWithAadhaarScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onContinue: (aadhaar: String, authMethod: String) -> Unit = { _, _ -> }
) {
    val colors = if (isDarkTheme) DarkSignInAadhaarColors else LightSignInAadhaarColors
    val typography = Ux4gTheme.typography

    var aadhaarNumber by rememberSaveable { mutableStateOf("") }
    var selectedMethod by rememberSaveable { mutableStateOf("otp") } // "otp" | "face"

    val digitsOnly = aadhaarNumber.replace(" ", "")
    val validationStatus = when {
        digitsOnly.length < 12 -> Ux4gInputFieldStatus.Default
        Ux4gAadhaarInputField.validateAadhaar(aadhaarNumber) -> Ux4gInputFieldStatus.Success
        else -> Ux4gInputFieldStatus.Error
    }
    val validationCaption = when (validationStatus) {
        Ux4gInputFieldStatus.Success -> "Aadhaar number looks valid"
        Ux4gInputFieldStatus.Error -> "Invalid Aadhaar number. Please check and re-enter."
        else -> null
    }

    val placeholderStyle = TextStyle(
        fontSize = 14.sp,
        fontWeight = FontWeight.Normal,
        color = colors.mutedText,
        lineHeight = 18.sp
    )

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 12.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(32.dp))

            Text(
                text = "Sign in with Aadhaar",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 28.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(6.dp))
            Text(
                text = "Enter your 12-digit Aadhaar number",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(24.dp))

            // Indian Aadhaar input with auto-formatting & Verhoeff validation
            Ux4gAadhaarInputField(
                value = aadhaarNumber,
                onValueChange = { aadhaarNumber = it },
                label = "Aadhaar Number",
                placeholder = "XXXX XXXX 1234",
                placeholderStyle = placeholderStyle,
                status = validationStatus,
                caption = validationCaption,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(20.dp))

            Text(
                text = "Choose Authentication Method",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(10.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                Box(modifier = Modifier.weight(1f)) {
                    Ux4gRadioButton(
                        value = "otp",
                        groupValue = selectedMethod,
                        onValueChange = { selectedMethod = it },
                        label = "Send OTP",
                        trailingIcon = Icons.Outlined.Phone
                    )
                }
                Box(modifier = Modifier.weight(1f)) {
                    Ux4gRadioButton(
                        value = "face",
                        groupValue = selectedMethod,
                        onValueChange = { selectedMethod = it },
                        label = "Face Auth",
                        trailingIcon = Icons.Outlined.Face
                    )
                }
            }
            Spacer(Modifier.height(24.dp))

            Ux4gButton(
                text = "Continue",
                onClick = { onContinue(aadhaarNumber, selectedMethod) },
                enabled = validationStatus == Ux4gInputFieldStatus.Success,
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(12.dp))

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Icon(
                    imageVector = Icons.Outlined.Lock,
                    contentDescription = null,
                    tint = colors.mutedText,
                    modifier = Modifier.size(14.dp)
                )
                Spacer(Modifier.width(6.dp))
                Text(
                    text = "Your Aadhaar details are encrypted and secure",
                    style = TextStyle(
                        fontSize = 12.sp,
                        color = colors.subtleText
                    )
                )
            }
        }

        BrandFooter(colors = colors)
    }
}

@Composable
private fun BackButton(onClick: () -> Unit) {
    Ux4gButton(
        text = "Back",
        onClick = onClick,
        variant = Ux4gButtonVariant.Ghost,
        size = Ux4gButtonSize.Small,
        leadingIcon = {
            Icon(
                imageVector = Icons.AutoMirrored.Filled.ArrowBack,
                contentDescription = "Back",
                modifier = Modifier.size(18.dp)
            )
        },
        modifier = Modifier.height(48.dp)
    )
}

@Composable
private fun BrandHeader(colors: SignInAadhaarColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16,
            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}

@Composable
private fun BrandFooter(colors: SignInAadhaarColors) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(top = Ux4gSpace.Space8, bottom = Ux4gSpace.Space20),
        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = "Powered by -",
            style = Ux4gTheme.typography.lS_default.copy(color = colors.mutedText)
        )
        Spacer(Modifier.width(Ux4gSpace.Space8))
        Image(
            painter = painterResource(AppR.drawable.ic_digital_india),
            contentDescription = "Digital India",
            colorFilter = colors.footerLogoTint?.let { ColorFilter.tint(it) },
            modifier = Modifier.height(22.dp)
        )
    }
}`}
