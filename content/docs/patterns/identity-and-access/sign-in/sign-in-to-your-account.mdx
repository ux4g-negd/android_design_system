# Sign in to your account

A government-grade sign-in pattern for citizen services. The screen combines a branded government header, Username and Password inputs, an inline error banner with an attempt counter, a primary **Send OTP** action, an **OR** separator, an **Aadhaar** sign-in fallback, and a registration link.

@Composable
fun SignInToYourAccountScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    errorMessage: String? = "Username not found.",
    attempt: Int = 1,
    maxAttempts: Int = 5,
    onMenuClick: () -> Unit = {},
    onSendOtp: (username: String, password: String) -> Unit = { _, _ -> },
    onSignInWithAadhaar: () -> Unit = {},
    onRegisterClick: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkSignInColors else LightSignInColors
    val typography = Ux4gTheme.typography

    var username by rememberSaveable { mutableStateOf("") }
    var password by rememberSaveable { mutableStateOf("") }

    val placeholderStyle = TextStyle(
        fontSize = 14.sp,
        fontWeight = FontWeight.Normal,
        color = colors.mutedText,
        lineHeight = 18.sp
    )

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = Ux4gSpace.Space20, end = Ux4gSpace.Space20, top = Ux4gSpace.Space24)
        ) {
            Text(
                text = "Sign in to your account",
                style = typography.hM_strong.copy(
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(6.dp))
            Text(
                text = "Access your government services securely",
                style = typography.bM_default.copy(color = colors.subtleText)
            )
            Spacer(Modifier.height(Ux4gSpace.Space24))

            Ux4gInputField(
                value = username,
                onValueChange = { username = it },
                label = "Username",
                placeholder = "Enter your username",
                placeholderStyle = placeholderStyle,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Ux4gInputField(
                value = password,
                onValueChange = { password = it },
                label = "Password",
                placeholder = "...........",
                placeholderStyle = placeholderStyle,
                type = Ux4gInputFieldType.Password,
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            if (errorMessage != null) {
                Ux4gStatusBanner(
                    variant = Ux4gBannerVariant.ErrorLight,
                    title = errorMessage,
                    titleStyle = TextStyle(
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Normal,
                        color = colors.errorText,
                        lineHeight = 18.sp
                    ),
                    subtitleWidget = {
                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Text(
                                text = "Take action",
                                style = TextStyle(
                                    fontSize = 14.sp,
                                    fontWeight = FontWeight.Bold,
                                    color = colors.errorText,
                                    lineHeight = 18.sp
                                ),
                                modifier = Modifier.weight(1f)
                            )
                            Text(
                                text = "Attempt $attempt of $maxAttempts",
                                style = typography.lM_default.copy(color = colors.errorText),
                                modifier = Modifier
                                    .clip(RoundedCornerShape(6.dp))
                                    .background(colors.attemptPillBackground)
                                    .padding(horizontal = 10.dp, vertical = 5.dp)
                            )
                        }
                    },
                    leadingIcon = {
                        Icon(
                            imageVector = Icons.Filled.Error,
                            contentDescription = null,
                            tint = colors.errorIcon,
                            modifier = Modifier.size(20.dp)
                        )
                    },
                    backgroundColor = colors.errorBackground,
                    borderColor = colors.errorBorder,
                    margin = PaddingValues(0.dp),
                    padding = PaddingValues(start = 12.dp, top = 12.dp, end = 10.dp, bottom = 12.dp)
                )
                Spacer(Modifier.height(Ux4gSpace.Space20))
            }

            Ux4gButton(
                text = "Send OTP",
                onClick = { onSendOtp(username, password) },
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Ux4gDivider(
                color = colors.border,
                labelSpacing = Ux4gSpace.Space16,
                label = {
                    Text(
                        text = "OR",
                        style = typography.lM_default.copy(
                            color = colors.mutedText,
                            letterSpacing = 0.5.sp
                        )
                    )
                }
            )
            Spacer(Modifier.height(Ux4gSpace.Space16))

            Ux4gButton(
                text = "Sign in with Aadhaar",
                onClick = onSignInWithAadhaar,
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Large,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )
            Spacer(Modifier.height(28.dp))

            Box(modifier = Modifier.fillMaxWidth(), contentAlignment = Alignment.Center) {
                Ux4gButton(
                    text = "New user? Register here",
                    onClick = onRegisterClick,
                    variant = Ux4gButtonVariant.Text,
                    size = Ux4gButtonSize.Medium,
                    modifier = Modifier.height(48.dp)
                )
            }
        }

        BrandFooter(colors = colors)
    }
}

@Composable
private fun BrandHeader(colors: SignInColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16,
            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}

@Composable
private fun BrandFooter(colors: SignInColors) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(top = Ux4gSpace.Space8, bottom = Ux4gSpace.Space20),
        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = "Powered by -",
            style = Ux4gTheme.typography.lS_default.copy(color = colors.mutedText)
        )
        Spacer(Modifier.width(Ux4gSpace.Space8))
        Image(
            painter = painterResource(AppR.drawable.ic_digital_india),
            contentDescription = "Digital India",
            colorFilter = colors.footerLogoTint?.let { ColorFilter.tint(it) },
            modifier = Modifier.height(22.dp)
        )
    }
}`}
