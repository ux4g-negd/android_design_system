# Your session is expiring

A security modal dialog displayed when a citizen remains inactive during a sensitive session or authenticated workflow. Built on the design system's `Ux4gModal` component, the pattern protects user data and prevents unauthorized access across three distinct lifecycle states:

1. **Expiring (Early Warning)**: Purple lock badge, prominent countdown timer (*"04:47"*), session explanation, animated linear progress bar, and dual actions (**Stay signed in** / **Sign out now**).
2. **Expiring soon (Final-Minute Warning)**: Orange warning badge, rapid countdown timer (*"00:57"*), urgent sign-out notification, orange progress indicator, and dual actions.
3. **Session ended (Terminal State)**: Neutral clock badge, saved-progress confirmation (*"Your form progress has been saved. Sign in again to continue where you left off."*), and a single full-width primary action (**Sign in to continue**).

### 1. Expiring (Early Warning State)

Early warning dialog with purple lock badge, live multi-minute countdown timer, linear progress track, and dual action buttons utilizing `Ux4gModal`.

import androidx.compose.foundation.background
import androidx.compose.foundation.isSystemInDarkTheme
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.Column

import com.ux4g.designsystemv1.components.modal.Ux4gModal
import com.ux4g.designsystemv1.components.modal.Ux4gModalFooterAlign
import com.ux4g.designsystemv1.components.modal.Ux4gModalFooterButtons
import com.ux4g.designsystemv1.foundation.Ux4gColors
import kotlinx.coroutines.delay

@Composable
fun SessionExpiringDialog(
    isOpen: Boolean = true,
    totalSeconds: Int = 300,
    initialSecondsLeft: Int = 287,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    onStaySignedIn: () -> Unit = {},
    onSignOut: () -> Unit = {},
    onDismiss: () -> Unit = {}
) {
    if (!isOpen) return

    var secondsLeft by remember { mutableIntStateOf(initialSecondsLeft) }

    LaunchedEffect(Unit) {
        while (secondsLeft > 0) {
            delay(1000L)
            secondsLeft--
        }
    }

    val minutes = (secondsLeft / 60).toString().padStart(2, '0')
    val seconds = (secondsLeft % 60).toString().padStart(2, '0')
    val formattedTime = "$minutes:$seconds"
    val progress = if (totalSeconds > 0) (secondsLeft.toFloat() / totalSeconds).coerceIn(0f, 1f) else 0f

    val iconBgColor = if (isDarkTheme) Ux4gColors.primary800 else Ux4gColors.primary100
    val iconColor = if (isDarkTheme) Ux4gColors.primary300 else Ux4gColors.primary600
    val countColor = iconColor
    val progressColor = iconColor
    val progressTrack = if (isDarkTheme) Ux4gColors.neutral700 else Ux4gColors.neutral200
    val descColor = if (isDarkTheme) Ux4gColors.neutral300 else Ux4gColors.neutral700

    Ux4gModal(
        onDismiss = onDismiss,
        showHeader = false,
        showCloseButton = false,
        showDividers = false,
        showBody = true,
        bodyContent = {
            Column(
                modifier = Modifier.fillMaxWidth(),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                // Purple Lock badge icon
                Box(
                    modifier = Modifier
                        .size(56.dp)
                        .clip(CircleShape)
                        .background(iconBgColor),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        imageVector = Icons.Filled.Lock,
                        contentDescription = "Session Expiring",
                        tint = iconColor,
                        modifier = Modifier.size(28.dp)
                    )
                }
                Spacer(Modifier.height(20.dp))

                Text(
                    text = "Your session is expiring",
                    style = TextStyle(
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Bold,
                        color = if (isDarkTheme) Ux4gColors.neutral50 else Ux4gColors.neutral900
                    ),
                    textAlign = TextAlign.Center
                )
                Spacer(Modifier.height(8.dp))

                Text(
                    text = formattedTime,
                    style = TextStyle(
                        fontSize = 40.sp,
                        fontWeight = FontWeight.ExtraBold,
                        color = countColor,
                        letterSpacing = (-0.5).sp,
                        lineHeight = 44.sp
                    ),
                    textAlign = TextAlign.Center
                )
                Spacer(Modifier.height(16.dp))

                Text(
                    text = "You've been inactive for a while. For your security, we'll sign you out automatically.",
                    style = TextStyle(
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Normal,
                        color = descColor,
                        lineHeight = 20.sp
                    ),
                    textAlign = TextAlign.Center
                )
                Spacer(Modifier.height(20.dp))

                // Progress Bar
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(6.dp)
                        .clip(RoundedCornerShape(3.dp))
                        .background(progressTrack)
                ) {
                    Box(
                        modifier = Modifier
                            .fillMaxWidth(fraction = progress)
                            .fillMaxHeight()
                            .clip(RoundedCornerShape(3.dp))
                            .background(progressColor)
                    )
                }
            }
        },
        showFooter = true,
        footerButtons = Ux4gModalFooterButtons.TwoButtons,
        footerAlign = Ux4gModalFooterAlign.Split,
        primaryButtonText = "Stay signed in",
        secondaryButtonText = "Sign out now",
        onPrimaryClick = onStaySignedIn,
        onSecondaryClick = onSignOut
    )
}`}
>

  </div>
