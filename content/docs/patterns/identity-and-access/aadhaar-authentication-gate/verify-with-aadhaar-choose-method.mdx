# Verify with Aadhaar - choose method

A citizen authentication gateway pattern allowing users to choose their preferred Aadhaar identity verification method. The screen features a branded government app header, a back navigation button, clear privacy messaging (*"Choose how you want to authenticate. Your Aadhaar number is never stored."*), interactive method selection cards with `Ux4gRadioButton` (**Aadhaar OTP** and **Face Authentication**), paired **Cancel** and **Continue** actions, and a pinned Digital India brand footer.

@Composable
fun VerifyWithAadhaarChooseMethodScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    initialMethod: String = "otp",
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {},
    onCancel: () -> Unit = {},
    onContinue: (selectedMethod: String) -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkVerifyAadhaarMethodColors else LightVerifyAadhaarMethodColors
    var selectedMethod by rememberSaveable { mutableStateOf(initialMethod) }

    val methods = listOf(
        AadhaarAuthMethodOption(
            id = "otp",
            title = "Aadhaar OTP",
            description = "Receive a one-time password on your Aadhaar-linked mobile number"
        ),
        AadhaarAuthMethodOption(
            id = "face",
            title = "Face Authentication",
            description = "Verify identity using face recognition through Aadhaar FaceRD service"
        )
    )

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(start = 20.dp, top = 24.dp, end = 20.dp, bottom = 0.dp)
        ) {
            BackButton(onClick = onBackClick)
            Spacer(Modifier.height(24.dp))

            Text(
                text = "Verify with Aadhaar",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    lineHeight = 28.sp,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            Text(
                text = "Choose how you want to authenticate. Your Aadhaar number is never stored.",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
            Spacer(Modifier.height(28.dp))

            // Method selection cards
            methods.forEach { methodOption ->
                val isSelected = selectedMethod == methodOption.id
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(RoundedCornerShape(12.dp))
                        .background(if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg)
                        .border(
                            width = 1.dp,
                            color = if (isSelected) colors.optionSelectedBorder else colors.optionUnselectedBorder,
                            shape = RoundedCornerShape(12.dp)
                        )
                        .clickable { selectedMethod = methodOption.id }
                        .padding(16.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.Top
                    ) {
                        Ux4gRadioButton(
                            selected = isSelected,
                            onClick = { selectedMethod = methodOption.id }
                        )
                        Spacer(Modifier.width(12.dp))
                        Column(modifier = Modifier.weight(1f)) {
                            Text(
                                text = methodOption.title,
                                style = TextStyle(
                                    fontSize = 16.sp,
                                    fontWeight = FontWeight.SemiBold,
                                    color = colors.title,
                                    lineHeight = 20.sp
                                )
                            )
                            Spacer(Modifier.height(4.dp))
                            Text(
                                text = methodOption.description,
                                style = TextStyle(
                                    fontSize = 13.sp,
                                    fontWeight = FontWeight.Normal,
                                    color = colors.subtleText,
                                    lineHeight = 17.sp
                                )
                            )
                        }
                    }
                }
                Spacer(Modifier.height(16.dp))
            }
        }

        // Action buttons
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 20.dp, vertical = 12.dp)
        ) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(12.dp),
                verticalAlignment = Alignment.CenterVertically
            ) {
                Ux4gButton(
                    text = "Cancel",
                    onClick = onCancel,
                    variant = Ux4gButtonVariant.Ghost,
                    size = Ux4gButtonSize.Large,
                    modifier = Modifier.weight(1f).height(48.dp)
                )
                Ux4gButton(
                    text = "Continue",
                    onClick = { onContinue(selectedMethod) },
                    size = Ux4gButtonSize.Large,
                    modifier = Modifier.weight(1f).height(48.dp)
                )
            }
        }

        BrandFooter(colors = colors)
    }
}

}
