# Application sent success

A confirmation pattern presented to citizens immediately upon successful submission of a government form or service petition.

The screen provides unequivocal positive feedback via a prominent checkmark illustration, an affirmative title (*"Application sent!"*), explicit service-level agreement (SLA) expectations (*"We will review and contact you within 3 working days."*), and paired primary/secondary navigation actions enabling citizens to either track application progress or return to their citizen dashboard.

### 1. Default Style (Full Screen)

Full-screen completion pattern featuring the branded government header, centered success icon, affirmative messaging, status check and dashboard actions, and the Digital India footer.

/**
 * Application Sent Success - Full Screen Default Pattern
 *
 * @param onViewStatus Invoked when citizen navigates to view submitted application status
 * @param onBackToDashboard Invoked when citizen returns to the main citizen dashboard

@Composable
fun ApplicationSentScreen(
    onViewStatus: () -> Unit = {},
    onBackToDashboard: () -> Unit = {}
) {
    val isDark = isSystemInDarkTheme()

    val surfaceBg = if (isDark) Ux4gColors.Neutral950 else Color.White
    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Ux4gColors.Neutral900
    val textSecondary = if (isDark) Ux4gColors.Neutral300 else Ux4gColors.Neutral600
    val dividerColor = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)

    val successBg = if (isDark) Color(0xFF064E3B) else Color(0xFFDCFCE7)
    val successIconTint = if (isDark) Color(0xFF34D399) else Color(0xFF16A34A)

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(surfaceBg)
    ) {
        // Branded National App Header
        Ux4gAppHeader(
            variant = if (isDark) Ux4gAppHeaderVariant.Dark else Ux4gAppHeaderVariant.Light,
            title = "",
            actions = listOf(
                Ux4gAppHeaderAction(
                    icon = Icons.Default.Menu,
                    contentDescription = "Menu",
                    onClick = {}
                )
            )
        )

        Divider(color = dividerColor, thickness = 1.dp)

        // Centered Content
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .padding(horizontal = 24.dp),
            horizontalAlignment = Alignment.CenterVertically,
            verticalArrangement = Arrangement.Center
        ) {
            // Success Circular Badge
            Box(
                modifier = Modifier
                    .size(80.dp)
                    .clip(CircleShape)
                    .background(successBg),
                contentAlignment = Alignment.Center
            ) {
                Icon(
                    imageVector = Icons.Default.CheckCircle,
                    contentDescription = "Success",
                    tint = successIconTint,
                    modifier = Modifier.size(48.dp)
                )
            }

            Spacer(modifier = Modifier.height(32.dp))

            // Main Success Heading
            Text(
                text = "Application sent!",
                style = TextStyle(
                    fontSize = 28.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = textPrimary,
                    textAlign = TextAlign.Center
                )
            )

            Spacer(modifier = Modifier.height(16.dp))

            // Informational SLA Subtitle
            Text(
                text = "We will review and contact you within 3 working days.",
                style = TextStyle(
                    fontSize = 16.sp,
                    fontWeight = FontWeight.Normal,
                    color = textSecondary,
                    textAlign = TextAlign.Center,
                    lineHeight = 24.sp
                ),
                modifier = Modifier.fillMaxWidth(0.9f)
            )
        }

        // Actions and Footer
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 24.dp, vertical = 16.dp)
        ) {
            Ux4gButton(
                text = "View application status",
                variant = Ux4gButtonVariant.Primary,
                size = Ux4gButtonSize.Large,
                onClick = onViewStatus,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(12.dp))

            Ux4gButton(
                text = "Back to dashboard",
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Large,
                onClick = onBackToDashboard,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(24.dp))

            // Brand Footer
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "Powered by Digital India",
                    style = TextStyle(
                        fontSize = 11.sp,
                        color = Ux4gColors.Neutral400,
                        fontWeight = FontWeight.Medium
                    )
                )
            }
        }
    }
}

@Preview(showBackground = true, widthDp = 360, heightDp = 760)
@Composable
fun ApplicationSentPreview() {
    ApplicationSentScreen()
}`}
