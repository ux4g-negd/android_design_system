# Government form with multiple errors

A comprehensive intake error-handling pattern designed to assist citizens when multiple input fields fail validation simultaneously upon form submission.

The pattern features a global high-visibility error summary banner (`Ux4gStatusBanner`) positioned above the form fields (*"3 fields need your attention. Fix them to continue."*) with an interactive action link (*"Jump to first error"*), paired with granular field-level validation feedback (`Ux4gInputFieldStatus.Error`) on invalid mobile numbers, malformed email addresses, and incomplete Aadhaar credentials.

                checked = acceptTerms,

                label = "Accept terms and conditions",
                isRequired = true
            )

                text = "Continue",
                variant = Ux4gButtonVariant.Primary,
                size = Ux4gButtonSize.Large,
                onClick = onContinue,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(12.dp))

            Ux4gButton(
                text = "Back",
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Large,
                onClick = onBack,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(24.dp))

            // Brand Footer
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "Powered by Digital India",
                    style = TextStyle(
                        fontSize = 11.sp,
                        color = Ux4gColors.Neutral400,
                        fontWeight = FontWeight.Medium
                    )
                )
            }
        }
    }
}
