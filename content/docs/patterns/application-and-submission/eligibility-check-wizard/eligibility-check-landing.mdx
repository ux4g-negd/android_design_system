# Eligibility Check Landing

A guided intake and orientation pattern introducing citizens to an eligibility assessment questionnaire prior to committing to a formal government petition or entitlement application.

The landing screen mitigates citizen uncertainty and drop-off by clarifying the scope of the assessment upfront—stating the number of questions (*5 questions*), estimated completion time (*~2 minutes*), and confirming that no physical documents or file uploads are required at this stage. Dual advancement options empower the citizen to either begin the step-by-step verification wizard or directly skip ahead if they already understand statutory requirements.

                    fontWeight = FontWeight.ExtraBold,

                ),
                textAlign = TextAlign.Center,
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(14.dp))

            // Subtitle Guidance
            Text(
                text = "Answer a few quick questions to see if you qualify.",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = textSecondary,
                    lineHeight = 20.sp

                textAlign = TextAlign.Center,
                modifier = Modifier.fillMaxWidth()

                                .size(8.dp)
                                .clip(CircleShape)
                                .background(primaryColor)

                        Spacer(modifier = Modifier.width(12.dp))

                        Text(
                            text = bulletText,
                            style = TextStyle(

                                color = textSecondary,
                                fontWeight = FontWeight.Normal
                            )
                        )
                    }
                }
            }

            Spacer(modifier = Modifier.height(32.dp))
        }

        // Bottom Actions & Navigation
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 24.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Divider(
                thickness = 1.dp,
                color = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)
            )

            Spacer(modifier = Modifier.height(16.dp))

            Ux4gButton(
                text = "Start Eligibility Check",
                variant = Ux4gButtonVariant.Primary,
                size = Ux4gButtonSize.Large,
                onClick = onStartCheck,
                modifier = Modifier

            Ux4gButton(
                text = "Skip and Apply Directly",
                variant = Ux4gButtonVariant.Ghost,
                size = Ux4gButtonSize.Large,
                onClick = onSkipAndApply,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(16.dp))

            Text(
                text = "Already applied? Track your application →",
                style = TextStyle(
                    fontSize = 13.sp,
                    fontWeight = FontWeight.Medium,
                    color = textSecondary
                ),
                modifier = Modifier
                    .clickable { onTrackApplication() }
                    .padding(vertical = 4.dp)
