# Eligibility Question Step

A focused, single-question assessment step in an eligibility verification wizard designed to capture citizen qualifying parameters incrementally without cognitive overload.

The pattern presents one discrete eligibility rule per screen—complete with progress feedback (*Question 1 of 5*), contextual statutory guidance explaining *why* the information is required, and rich selectable option cards that accommodate both touch and keyboard accessibility. Clear primary and secondary actions guide the citizen forward or allow safe backtrack navigation without losing current selections.

            title = "Yes",
            subtitle = "I am a resident of Maharashtra"

            title = "No",

            variant = Ux4gAppHeaderVariant.Light,

                Ux4gAppHeaderAction(
                    icon = "menu",
                    onClick = {}
                )
            ),
            modifier = Modifier.fillMaxWidth()
        )

            thickness = 1.dp,
            color = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)
        )

        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(scrollState)
                .padding(horizontal = 20.dp, vertical = 20.dp)
        ) {
            // Step Progress Indicator
            Text(
                text = "Question $currentStep of $totalSteps",
                style = TextStyle(
                    fontSize = 13.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = textSecondary
                )
            )

            Spacer(modifier = Modifier.height(8.dp))

            // Progress Bar Track & Indicator
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(6.dp)
                    .clip(RoundedCornerShape(999.dp))
                    .background(trackBg)
            ) {
                Box(

                        .fillMaxWidth(fraction = progressFraction)
                        .height(6.dp)
                        .clip(RoundedCornerShape(999.dp))
                        .background(primaryColor)
                )

            Spacer(modifier = Modifier.height(24.dp))

            Text(
                text = "Are you a resident of Maharashtra?",

                    fontWeight = FontWeight.Bold,
                    color = textPrimary,
                    lineHeight = 28.sp

            )

            Spacer(modifier = Modifier.height(8.dp))

            // Helper Context
            Text(
                text = "You must be a resident to apply for a state-issued income certificate.",
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Normal,
                    color = textSecondary,
                    lineHeight = 20.sp
                )
            )

            Spacer(modifier = Modifier.height(24.dp))

            Column(
                verticalArrangement = Arrangement.spacedBy(12.dp),
                modifier = Modifier.fillMaxWidth()
            ) {
                options.forEach { option ->
                    val isSelected = selectedOption == option.id

                    val cardBg = when {
                        isSelected -> if (isDark) Ux4gColors.Primary900 else Ux4gColors.Primary50
                        else -> if (isDark) Ux4gColors.Neutral900 else Color.White
                    }

                    val borderColor = when {
                        isSelected -> if (isDark) Ux4gColors.Primary400 else Ux4gColors.Primary200
                        else -> if (isDark) Ux4gColors.Neutral700 else Color(0xFFE5E7EB)
                    }

                    val radioBorderColor = when {
                        isSelected -> if (isDark) Ux4gColors.Primary300 else Ux4gColors.Primary600
                        else -> if (isDark) Ux4gColors.Neutral700 else Ux4gColors.Neutral200
                    }

                    val titleColor = when {
                        isSelected -> if (isDark) Ux4gColors.Primary300 else Ux4gColors.Primary600
                        else -> textPrimary
                    }

                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .fillMaxWidth()
                            .clip(RoundedCornerShape(12.dp))
                            .background(cardBg)
                            .border(1.5.dp, borderColor, RoundedCornerShape(12.dp))
                            .clickable { selectedOption = option.id }
                            .padding(horizontal = 16.dp, vertical = 14.dp)
                    ) {
                        // Custom Accessible Radio Button Indicator
                        Box(
                            contentAlignment = Alignment.Center,
                            modifier = Modifier
                                .size(22.dp)
                                .clip(CircleShape)
                                .background(if (isDark) Ux4gColors.Neutral900 else Color.White)
                                .border(
                                    width = if (isSelected) 6.dp else 2.dp,
                                    color = radioBorderColor,
                                    shape = CircleShape
                                )
                        )

                        Spacer(modifier = Modifier.width(14.dp))

                        // Option Labels
                        Column(
                            modifier = Modifier.weight(1f)
                        ) {
                            Text(
                                text = option.title,
                                style = TextStyle(
                                    fontSize = 16.sp,
                                    fontWeight = FontWeight.Bold,
                                    color = titleColor
                                )
                            )

                            Spacer(modifier = Modifier.height(2.dp))

                            Text(
                                text = option.subtitle,
                                style = TextStyle(
                                    fontSize = 13.sp,
                                    color = textSecondary,
                                    lineHeight = 18.sp
                                )
                            )
                        }
                    }
                }
            }
        }

        // Bottom Actions Container
        Divider(
            thickness = 1.dp,
            color = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)

            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 20.dp, vertical = 16.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Ux4gButton(
                text = "Continue",
                variant = Ux4gButtonVariant.Primary,
                size = Ux4gButtonSize.Large,
                enabled = selectedOption.isNotEmpty(),
                onClick = { onContinue(selectedOption) },

                    .fillMaxWidth()

                style = TextStyle(
                    fontSize = 11.sp,
                    color = Ux4gColors.Neutral400,
                    fontWeight = FontWeight.Medium
                )

    }
