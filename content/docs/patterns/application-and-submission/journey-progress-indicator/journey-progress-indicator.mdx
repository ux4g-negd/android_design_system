# Journey Progress Indicator

A structured multi-step intake pattern providing citizens with continuous orientation and progression feedback across sequential stages of a complex public service petition.

The journey progress indicator uses a standardized horizontal stepper (`Ux4gStepper`) at the top of the workflow to visually represent overall progress (*Eligibility*, *Personal*, *Documents*, *Submit*). By clearly distinguishing between completed, active, and upcoming steps, citizens maintain a clear mental model of remaining requirements. The form body presents discrete data capture fields with inline validation, an explicit declaration confirmation, and predictable forward/back navigational controls.

### 1. Default Style (Full Screen)

Standard full-screen layout featuring a top government app header, horizontal step tracker, clean form inputs (full name, domain dropdown, PAN card input), declaration checkbox, and dual action buttons.

/**

 * @param onContinue Callback invoked when citizen advances to next stage
 * @param onBack Callback invoked when navigating back to previous step

@Composable
fun JourneyProgressIndicatorScreen(
    onContinue: () -> Unit = {},
    onBack: () -> Unit = {}
) {

    val scrollState = rememberScrollState()

    var fullName by remember { mutableStateOf("") }
    var selectedDomain by remember { mutableStateOf<String?>(null) }
    var panNumber by remember { mutableStateOf("ABCDE1234F") }
    var confirmDetails by remember { mutableStateOf(false) }

    val surfaceBg = if (isDark) Ux4gColors.Neutral950 else Color(0xFFFAFAFA)
    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Ux4gColors.Neutral900
    val textSecondary = if (isDark) Ux4gColors.Neutral400 else Ux4gColors.Neutral700

    val domainOptions = listOf(
        Ux4gDropdownOption(id = "agriculture", label = "Agriculture"),
        Ux4gDropdownOption(id = "education", label = "Education"),
        Ux4gDropdownOption(id = "finance", label = "Finance"),
        Ux4gDropdownOption(id = "healthcare", label = "Healthcare")
    )

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(surfaceBg)
    ) {
        // App Header
        Ux4gAppHeader(
            variant = Ux4gAppHeaderVariant.Light,
            leadingWidgets = listOf("emblem", "union"),
            actions = listOf(
                Ux4gAppHeaderAction(
                    icon = "menu",
                    onClick = {}
                )
            ),
            modifier = Modifier.fillMaxWidth()
        )

        Divider(
            thickness = 1.dp,
            color = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)
        )

        // Scrollable Body Content
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(scrollState)
                .padding(horizontal = 20.dp, vertical = 20.dp)
        ) {
            // Stepper Progression Indicator
            Ux4gStepper(
                totalSteps = 4,
                currentStep = 3,
                stepSize = 20.dp,
                steps = listOf(
                    Ux4gStepItem(title = "Eligibility"),
                    Ux4gStepItem(title = "Personal"),
                    Ux4gStepItem(title = "Documents"),
                    Ux4gStepItem(title = "Submit")
                ),
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(28.dp))

            // Section Heading & Subtitle
            Text(
                text = "Verify your details",
                style = TextStyle(
                    fontSize = 22.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = textPrimary,
                    lineHeight = 28.sp
                )
            )

            Spacer(modifier = Modifier.height(6.dp))

            Text(
                text = "Confirm the information below to proceed.",
                style = TextStyle(
                    fontSize = 13.sp,
                    color = textSecondary,
                    lineHeight = 18.sp
                )
            )

            Spacer(modifier = Modifier.height(20.dp))

            // Full Name Input
            Ux4gInputField(
                value = fullName,
                onValueChange = { fullName = it },
                label = "Full name (as per Aadhaar)",
                placeholder = "Placeholder",
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(16.dp))

            // Domain Selection Dropdown
            Ux4gSelectionDropdown(
                options = domainOptions,
                selectedOptionIds = selectedDomain?.let { listOf(it) } ?: emptyList(),
                onSelectionChange = { selected ->
                    selectedDomain = selected.firstOrNull()
                },
                label = "Select domain",
                placeholder = "Please select..",
                mode = Ux4gDropdownMode.Single,
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(16.dp))

            // PAN Input Field
            Ux4gPanInputField(
                value = panNumber,
                onValueChange = { panNumber = it },
                label = "PAN number",
                placeholder = "ABCDE1234F",
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(20.dp))

            // Confirmation Checkbox
            Ux4gCheckbox(
                value = confirmDetails,
                onValueChange = { confirmDetails = it },
                label = "I confirm these details match my official documents.",
                isRequired = true,
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(28.dp))

            // Actions
            Ux4gButton(
                text = "Continue",
                variant = Ux4gButtonVariant.Primary,
                size = Ux4gButtonSize.Large,
                enabled = confirmDetails,
                onClick = onContinue,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(12.dp))

            Ux4gButton(
                text = "Back",
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Large,
                onClick = onBack,
                modifier = Modifier
                    .fillMaxWidth()
                    .height(48.dp)
            )

            Spacer(modifier = Modifier.height(20.dp))

            // Footer
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Text(
                    text = "Powered by Digital India",
                    style = TextStyle(
                        fontSize = 11.sp,
                        color = Ux4gColors.Neutral400,
                        fontWeight = FontWeight.Medium
                    )
                )
            }
        }
    }
}
