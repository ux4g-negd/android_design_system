# Unsaved Changes Dialog

A protective modal dialogue built with `Ux4gModal` that intercepts navigation when a citizen attempts to leave a multi-step form or portal workflow with uncommitted changes.

The dialog alerts the citizen that leaving immediately will discard modifications made since their last save checkpoint. It presents two explicit pathways: **Save draft and leave** to commit pending edits and safely exit, or **Leave without saving** for situations where the applicant intentionally wishes to abandon current changes.

import com.ux4g.designsystemv1.components.modal.Ux4gModal
import com.ux4g.designsystemv1.foundation.Ux4gColors

/**
 * Unsaved Changes Dialog - Jetpack Compose Pattern using Ux4gModal
 *
 * @param isOpen Whether the dialog is currently visible
 * @param onSaveAndLeave Callback when saving draft and exiting
 * @param onLeaveWithoutSaving Callback when discarding changes and exiting
 * @param onDismiss Callback when dismissing the dialog
 */
@Composable
fun UnsavedChangesDialog(
    isOpen: Boolean = true,
    onSaveAndLeave: () -> Unit = {},
    onLeaveWithoutSaving: () -> Unit = {},
    onDismiss: () -> Unit = {}
) {
    if (!isOpen) return

    val isDark = isSystemInDarkTheme()

    val dialogBg = if (isDark) Ux4gColors.Neutral900 else Color.White
    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Color(0xFF111827)
    val textSecondary = if (isDark) Ux4gColors.Neutral400 else Color(0xFF4B5563)
    val discardColor = if (isDark) Ux4gColors.Red300 else Ux4gColors.Red800

    Ux4gModal(
        onDismiss = onDismiss,
        showHeader = false,
        showCloseButton = false,
        showDividers = false,
        showBody = true,
        showFooter = false,
        backgroundColor = dialogBg,
        cornerRadius = 16.dp,
        bodyContent = {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(8.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                // Dialog Title
                Text(
                    text = "Unsaved changes",
                    style = TextStyle(
                        fontSize = 20.sp,
                        fontWeight = FontWeight.Bold,
                        color = textPrimary
                    ),
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(12.dp))

                // Description
                Text(
                    text = "If you leave now, changes since your last save will be lost.",
                    style = TextStyle(
                        fontSize = 14.sp,
                        color = textSecondary,
                        lineHeight = 20.sp
                    ),
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(24.dp))

                // Primary Action: Save draft and leave
                Ux4gButton(
                    text = "Save draft and leave",
                    variant = Ux4gButtonVariant.Primary,
                    size = Ux4gButtonSize.Large,
                    onClick = onSaveAndLeave,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                )

                Spacer(modifier = Modifier.height(12.dp))

                // Secondary Action: Leave without saving
                Box(
                    modifier = Modifier
                        .clip(RoundedCornerShape(8.dp))
                        .clickable { onLeaveWithoutSaving() }
                        .padding(vertical = 8.dp, horizontal = 12.dp),
                    contentAlignment = Alignment.Center
                ) {
                    Text(
                        text = "Leave without saving",
                        style = TextStyle(
                            fontSize = 14.sp,
                            fontWeight = FontWeight.Medium,
                            color = discardColor
                        )
                    )
                }
            }
        }
    )
}

@Preview(showBackground = true, widthDp = 360, heightDp = 760)
@Composable
fun UnsavedChangesDialogPreview() {
    UnsavedChangesDialog()
}`}
