# Discard Draft Dialog

A high-consequence confirmation modal built with `Ux4gModal` that intercepts citizen requests to discard active application drafts.

Because discarding an in-progress citizen application permanently purges entered data and uploaded files, the pattern emphasizes the irreversible nature of the action (*"Your saved progress will be deleted. This cannot be undone."*). It pairs an explicit destructive primary trigger (**Discard and start again**) with a prominent safe retention action (**Keep my draft**) to prevent accidental loss of extensive work.

/**
 * Discard Draft Dialog - Jetpack Compose Pattern using Ux4gModal
 *
 * @param isOpen Whether the confirmation dialog is currently visible
 * @param onDiscardAndStartAgain Callback when permanently discarding the draft
 * @param onKeepDraft Callback when retaining the draft and closing the dialog
 * @param onDismiss Callback when dismissing the dialog by tapping outside
 */
@Composable
fun DiscardDraftDialog(
    isOpen: Boolean = true,
    onDiscardAndStartAgain: () -> Unit = {},
    onKeepDraft: () -> Unit = {},
    onDismiss: () -> Unit = {}
) {
    if (!isOpen) return

    val isDark = isSystemInDarkTheme()

    val dialogBg = if (isDark) Ux4gColors.Neutral900 else Color.White
    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Color(0xFF111827)
    val textSecondary = if (isDark) Ux4gColors.Neutral400 else Color(0xFF4B5563)
    val destructiveButtonBg = if (isDark) Ux4gColors.Red500 else Ux4gColors.Red600

    Ux4gModal(
        onDismiss = onDismiss,
        showHeader = false,
        showCloseButton = false,
        showDividers = false,
        showBody = true,
        showFooter = false,
        backgroundColor = dialogBg,
        cornerRadius = 16.dp,
        bodyContent = {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(8.dp),
                horizontalAlignment = Alignment.CenterHorizontally
            ) {
                // Title
                Text(
                    text = "Discard draft?",
                    style = TextStyle(
                        fontSize = 20.sp,
                        fontWeight = FontWeight.Bold,
                        color = textPrimary
                    ),
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(12.dp))

                // Description
                Text(
                    text = "Your saved progress will be deleted.\\nThis cannot be undone.",
                    style = TextStyle(
                        fontSize = 14.sp,
                        color = textSecondary,
                        lineHeight = 20.sp
                    ),
                    textAlign = TextAlign.Center
                )

                Spacer(modifier = Modifier.height(24.dp))

                // Destructive Primary Action
                Ux4gButton(
                    text = "Discard and start again",
                    variant = Ux4gButtonVariant.Primary,
                    size = Ux4gButtonSize.Large,
                    onClick = onDiscardAndStartAgain,

                        .fillMaxWidth()
                        .height(48.dp)
                )

                Spacer(modifier = Modifier.height(12.dp))

                // Safe Secondary Action: Keep Draft
                Ux4gButton(
                    text = "Keep my draft",
                    variant = Ux4gButtonVariant.Outline,
                    size = Ux4gButtonSize.Large,
                    onClick = onKeepDraft,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                )
            }
        }
    )
}
