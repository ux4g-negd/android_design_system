# Payment Processing

A secure payment processing and gateway redirection pattern for citizen-facing digital services. The screen represents the active payment stage in a multi-step public service application flow. It presents the citizen with available payment methods (UPI, Net Banking, Debit / Credit Card, and CSC Centre cash payments) while displaying an immediate focus-trapping redirection modal dialog powered by `Ux4gModal` and `Ux4gLoader`. This assures citizens of secure transaction transmission to the government payment gateway (PayGov / Treasury) and explicitly warns against closing or refreshing the screen.

@Composable
fun PaymentProcessingScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    amountText: String = "Rs 41.30",

        )

        listOf(
            Ux4gStepItem(title = "Personal info", statusLabel = "Completed"),
            Ux4gStepItem(title = "Documents", statusLabel = "Completed"),
            Ux4gStepItem(title = "Review", statusLabel = "Completed"),
            Ux4gStepItem(title = "Payment", statusLabel = "In progress")
        )
    }

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        // Background Form Layout
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
        ) {
            PaymentBrandHeader(colors = colors, onMenuClick = onMenuClick)

            Ux4gDivider(color = colors.border)

            // Horizontal Stepper
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = Ux4gSpace.Space16, vertical = Ux4gSpace.Space10)
            ) {
                Ux4gStepper(
                    totalSteps = 4,
                    currentStep = 4,
                    steps = steps,
                    orientation = StepperOrientation.Horizontal,
                    linePlacement = StepperLinePlacement.Center

            }

            // Main Content Area
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = Ux4gSpace.Space16, vertical = Ux4gSpace.Space12)
            ) {
                Text(
                    text = "Choose payment method",
                    style = TextStyle(
                        fontSize = 22.sp,
                        fontWeight = FontWeight.ExtraBold,
                        color = colors.title,

                        isSelected = isSelected,
                        colors = colors,

        }
    }

@Composable
private fun PaymentOptionTile(
    item: PaymentOptionItem,
    isSelected: Boolean,
    colors: PaymentProcessingColors,
    onSelect: () -> Unit
) {
    val bg = if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg
    val titleColor = if (isSelected) colors.optionSelectedTitle else colors.optionUnselectedTitle
    val subColor = if (isSelected) colors.optionSelectedSub else colors.optionUnselectedSub

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(bg)
            .clickable(onClick = onSelect)
            .padding(horizontal = 16.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Ux4gRadioButton(
            value = item.id,
            groupValue = if (isSelected) item.id else "",
            onValueChange = { onSelect() },
            size = Ux4gRadioButtonSize.Medium
        )
        Spacer(Modifier.width(12.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = item.title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = titleColor
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = item.subtitle,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}

@Composable
private fun PaymentOptionTileCard(
    item: PaymentOptionItem,
    isSelected: Boolean,
    colors: PaymentProcessingCardColors,
    onSelect: () -> Unit
) {
    val bg = if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg
    val titleColor = if (isSelected) colors.optionSelectedTitle else colors.optionUnselectedTitle
    val subColor = if (isSelected) colors.optionSelectedSub else colors.optionUnselectedSub

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(bg)
            .clickable(onClick = onSelect)
            .padding(horizontal = 16.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Ux4gRadioButton(
            value = item.id,
            groupValue = if (isSelected) item.id else "",
            onValueChange = { onSelect() },
            size = Ux4gRadioButtonSize.Medium
        )
        Spacer(Modifier.width(12.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = item.title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = titleColor
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = item.subtitle,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}
