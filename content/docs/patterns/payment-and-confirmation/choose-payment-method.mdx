# Choose Payment Method

A core payment channel selection pattern for citizen-facing digital public services. When an application or citizen filing reaches the statutory payment milestone (e.g., certificate issuance fees, building permit charges, trade licenses, or municipal taxes), this pattern provides a clear, transparent choice among standard national payment channels.

It supports:
- **UPI (Unified Payments Interface)**: Mobile-first instant payments with inline UPI ID field verification.
- **Net Banking**: Direct debit authorization across 60+ Indian commercial and public sector banks.
- **Debit / Credit Card**: RuPay, Visa, and MasterCard transaction processing.
- **CSC Centre**: Cash-on-counter authorization at village-level Common Service Centres (ensuring vital access for rural and unbanked citizens).

/**
 * Full-screen Choose Payment Method pattern.
 *
 * @param amountText Payable statutory amount (e.g. "Rs 41.30").
 * @param onPayClick Callback invoked when user triggers the primary pay action.
 * @param onBackClick Callback invoked when user taps "Back to review".
 * @param onMenuClick Callback triggered for top header menu navigation.

@Composable
fun ChoosePaymentMethodScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    amountText: String = "Rs 41.30",
    onPayClick: (method: String, upiId: String) -> Unit = { _, _ -> },
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {}
) {

            PaymentMethodOption("card", "Debit / Credit Card", "Visa, MasterCard, RuPay"),
            PaymentMethodOption("csc", "CSC Centre", "Pay in cash at your nearest CSC")
        )
    }

    val steps = remember {
        listOf(
            Ux4gStepItem(title = "Personal info", statusLabel = "Completed"),
            Ux4gStepItem(title = "Documents", statusLabel = "Completed"),
            Ux4gStepItem(title = "Review", statusLabel = "Completed"),
            Ux4gStepItem(title = "Payment", statusLabel = "In progress")
        )
    }

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {

            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())

                    orientation = StepperOrientation.Horizontal,
                    linePlacement = StepperLinePlacement.Center
                )
            }

            // Main Content Area
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = Ux4gSpace.Space16, vertical = Ux4gSpace.Space12)
            ) {
                Text(
                    text = "Choose payment method",
                    style = TextStyle(

                        fontWeight = FontWeight.ExtraBold,
                        color = colors.title,
                        letterSpacing = (-0.3).sp,
                        lineHeight = 28.sp
                    )
                )
                Spacer(Modifier.height(4.dp))
                Text(
                    text = "Select how you would like to pay $amountText.",
                    style = TextStyle(
                        fontSize = 13.sp,
                        color = colors.subtleText,
                        lineHeight = 18.sp
                    )
                )

                        item = option,
                        isSelected = isSelected,
                        colors = colors,
                        onSelect = { selectedMethod = option.id }
                    )
                    Spacer(Modifier.height(Ux4gSpace.Space8))
                }

                // Dynamic UPI ID Input Field
                if (selectedMethod == "upi") {
                    Spacer(Modifier.height(Ux4gSpace.Space8))

                        value = upiId,

                    text = "Pay $amountText",

                    size = Ux4gButtonSize.Large,

                        size = Ux4gButtonSize.Medium,
                        modifier = Modifier.height(44.dp)
                    )
                }
            }

            Spacer(Modifier.weight(1f, fill = false))
            PaymentBrandFooter(colors = colors)
        }
    }
}

@Composable
private fun PaymentMethodOptionTile(
    item: PaymentMethodOption,
    isSelected: Boolean,
    colors: ChoosePaymentMethodColors,
    onSelect: () -> Unit
) {
    val bg = if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg
    val titleColor = if (isSelected) colors.optionSelectedTitle else colors.optionUnselectedTitle
    val subColor = if (isSelected) colors.optionSelectedSub else colors.optionUnselectedSub

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(bg)
            .clickable(onClick = onSelect)
            .padding(horizontal = 16.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Ux4gRadioButton(
            value = item.id,
            groupValue = if (isSelected) item.id else "",
            onValueChange = { onSelect() },
            size = Ux4gRadioButtonSize.Medium
        )
        Spacer(Modifier.width(12.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = item.title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = titleColor
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = item.subtitle,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}

                        .border(1.dp, colors.cardBorder, RoundedCornerShape(16.dp))
                        .padding(horizontal = Ux4gSpace.Space16, vertical = Ux4gSpace.Space20)
                ) {
                    Column(modifier = Modifier.fillMaxWidth()) {
                        Text(
                            text = "Choose payment method",
                            style = TextStyle(
                                fontSize = 22.sp,
                                fontWeight = FontWeight.ExtraBold,
                                color = colors.title,
                                letterSpacing = (-0.3).sp,
                                lineHeight = 28.sp
                            )
                        )
                        Spacer(Modifier.height(4.dp))
                        Text(
                            text = "Select how you would like to pay $amountText.",
                            style = TextStyle(
                                fontSize = 13.sp,
                                color = colors.subtleText,
                                lineHeight = 18.sp
                            )
                        )

                        Spacer(Modifier.height(Ux4gSpace.Space20))

                        // Payment Options Radio List
                        paymentOptions.forEach { option ->
                            val isSelected = option.id == selectedMethod
                            PaymentMethodOptionTileCard(
                                item = option,
                                isSelected = isSelected,
                                colors = colors,
                                onSelect = { selectedMethod = option.id }
                            )
                            Spacer(Modifier.height(Ux4gSpace.Space8))
                        }

                        // Dynamic UPI ID Input Field
                        if (selectedMethod == "upi") {
                            Spacer(Modifier.height(Ux4gSpace.Space8))
                            Ux4gInputField(
                                value = upiId,
                                onValueChange = { upiId = it },
                                label = "Enter UPI ID",
                                placeholder = "e.g. name@upi",
                                modifier = Modifier.fillMaxWidth()
                            )
                            Spacer(Modifier.height(Ux4gSpace.Space16))
                        }

                        Spacer(Modifier.height(Ux4gSpace.Space16))

                        // Action Buttons
                        Ux4gButton(
                            text = "Pay $amountText",
                            onClick = { onPayClick(selectedMethod, upiId) },
                            variant = Ux4gButtonVariant.Primary,
                            size = Ux4gButtonSize.Large,
                            modifier = Modifier
                                .fillMaxWidth()
                                .height(48.dp)
                        )
                        Spacer(Modifier.height(Ux4gSpace.Space12))

                        Box(
                            modifier = Modifier.fillMaxWidth(),
                            contentAlignment = Alignment.Center
                        ) {
                            Ux4gButton(
                                text = "Back to review",
                                onClick = onBackClick,
                                variant = Ux4gButtonVariant.Ghost,
                                size = Ux4gButtonSize.Medium,
                                modifier = Modifier.height(44.dp)
                            )
                        }
                    }
                }
            }

            Spacer(Modifier.weight(1f, fill = false))
            PaymentCardBrandFooter(colors = colors)
        }
    }

@Composable
private fun PaymentMethodOptionTileCard(
    item: PaymentMethodOption,
    isSelected: Boolean,
    colors: ChoosePaymentMethodCardColors,
    onSelect: () -> Unit
) {
    val bg = if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg
    val titleColor = if (isSelected) colors.optionSelectedTitle else colors.optionUnselectedTitle
    val subColor = if (isSelected) colors.optionSelectedSub else colors.optionUnselectedSub

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(bg)
            .clickable(onClick = onSelect)
            .padding(horizontal = 16.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Ux4gRadioButton(
            value = item.id,
            groupValue = if (isSelected) item.id else "",
            onValueChange = { onSelect() },
            size = Ux4gRadioButtonSize.Medium
        )
        Spacer(Modifier.width(12.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = item.title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = titleColor
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = item.subtitle,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}
