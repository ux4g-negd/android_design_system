# Payment Successful

A positive confirmation pattern presented to citizens following a successful digital fee or service payment. Built with trusted government design standards, the screen displays a high-visibility green status badge, clear payment confirmation copy, a structured transaction receipt audit card (amount, transaction ID, payment mode, and timestamp), primary application tracking action, a PDF receipt download CTA, and official Digital India branding.

@Composable
fun PaymentSuccessfulScreen(
    modifier: Modifier = Modifier,

    amountText: String = "Rs 41.30",
    transactionId: String = "PG2026MH04127TX",
    paymentMethod: String = "UPI · ramesh@upi",

                        imageVector = Icons.Filled.Check,
                        contentDescription = "Success",
                        tint = Color.White,
                        modifier = Modifier.size(22.dp)
                    )
                }
            }

            Spacer(Modifier.height(Ux4gSpace.Space16))

            Text(
                text = "Payment Successful",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = colors.title,
                    letterSpacing = (-0.3).sp,
                    lineHeight = 28.sp
                ),
                textAlign = TextAlign.Center

            Spacer(Modifier.height(6.dp))

                text = "$amountText paid via UPI to $departmentName.",

                textAlign = TextAlign.Center
            )

@Composable
private fun ReceiptDetailsCard(
    amount: String,
    transactionId: String,
    paymentMethod: String,
    timestamp: String,
    colors: PaymentSuccessColors
) {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(colors.receiptCardBg)
            .border(1.dp, colors.receiptCardBorder, RoundedCornerShape(12.dp))
            .padding(horizontal = 16.dp, vertical = 14.dp)
    ) {
        Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
            ReceiptRow(label = "Amount paid", value = amount, isBold = true, colors = colors)
            ReceiptRow(label = "Transaction ID", value = transactionId, isBold = true, colors = colors)
            ReceiptRow(label = "Method", value = paymentMethod, isBold = true, colors = colors)
            ReceiptRow(label = "Date and time", value = timestamp, isBold = true, colors = colors)
        }
    }
}

@Composable
private fun ReceiptRow(
    label: String,
    value: String,
    isBold: Boolean,
    colors: PaymentSuccessColors
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = label,
            style = TextStyle(
                fontSize = 13.sp,
                fontWeight = FontWeight.Normal,
                color = colors.receiptLabelText
            )
        )
        Text(
            text = value,
            style = TextStyle(
                fontSize = 13.sp,
                fontWeight = if (isBold) FontWeight.Bold else FontWeight.SemiBold,
                color = colors.receiptValueText,
                textAlign = TextAlign.End
            )
        )
    }
}
