# Payment Failed

A critical error and recovery pattern presented to citizens when an online payment gateway declines or fails to complete a transaction. Designed to prevent panic and confusion, the screen provides immediate reassurance that no funds were deducted, presents a transparent failure audit card with the bank decline reason and transaction timestamp, offers an immediate retry option, and provides alternative payment methods (including offline CSC Centre payments).

@Composable
private fun FailureDetailsCard(
    reason: String,
    attemptedAmount: String,
    paymentMethod: String,
    timestamp: String,
    colors: PaymentFailedColors
) {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(colors.failureCardBg)
            .border(1.dp, colors.failureCardBorder, RoundedCornerShape(12.dp))
            .padding(horizontal = 16.dp, vertical = 14.dp)
    ) {
        Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
            FailureRow(label = "Reason", value = reason, isBold = true, colors = colors)
            FailureRow(label = "Attempted amount", value = attemptedAmount, isBold = true, colors = colors)
            FailureRow(label = "Method", value = paymentMethod, isBold = true, colors = colors)
            FailureRow(label = "Date and time", value = timestamp, isBold = true, colors = colors)
        }
    }
}

@Composable

    label: String,
    value: String,
    isBold: Boolean,
    colors: PaymentFailedColors
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        horizontalArrangement = Arrangement.SpaceBetween,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = label,
            style = TextStyle(
                fontSize = 13.sp,
                fontWeight = FontWeight.Normal,
                color = colors.failureLabelText
            )
        )
        Text(
            text = value,
            style = TextStyle(
                fontSize = 13.sp,
                fontWeight = if (isBold) FontWeight.Bold else FontWeight.SemiBold,
                color = colors.failureValueText,
                textAlign = TextAlign.End
            )
        )
    }
}
