# Payment

A secure payment method selection pattern for citizen-facing digital public services. This screen represents the payment stage in a multi-step public service workflow (such as certificate issuances, municipal tax collection, application fees, or trade licenses).

It allows citizens to choose among standard national payment channels—**UPI**, **Net Banking**, **Debit / Credit Card**, and **CSC Centre cash payments** (ensuring access for non-digitally enabled or rural citizens). The screen provides explicit statutory security guarantees via an SSL encryption notice, an active multi-step stepper, and clear forward/backward journey progression.

/**
 * Default Full Screen Payment selection pattern.
 *
 * @param amountText Total fee payable (e.g. "Rs 41.30").
 * @param onPayClick Callback triggered when user taps the primary pay action.
 * @param onBackClick Callback triggered when user navigates back to the review step.
 * @param onMenuClick Callback triggered for top navigation menu icon.
 */
@Composable
fun PaymentScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    amountText: String = "Rs 41.30",
    onPayClick: (method: String, upiId: String) -> Unit = { _, _ -> },
    onBackClick: () -> Unit = {},
    onMenuClick: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkPaymentColors else LightPaymentColors

    var selectedMethod by remember { mutableStateOf("upi") }
    var upiId by remember { mutableStateOf("") }

    val paymentOptions = remember {
        listOf(
            PaymentOptionItem("upi", "UPI", "Pay with any UPI app"),
            PaymentOptionItem("netbanking", "Net Banking", "60+ banks supported"),
            PaymentOptionItem("card", "Debit / Credit Card", "Visa, MasterCard, RuPay"),
            PaymentOptionItem("csc", "CSC Centre", "Pay in cash at your nearest CSC")
        )
    }

    val steps = remember {
        listOf(
            Ux4gStepItem(title = "Personal info", statusLabel = "Completed"),
            Ux4gStepItem(title = "Documents", statusLabel = "Completed"),
            Ux4gStepItem(title = "Review", statusLabel = "Completed"),
            Ux4gStepItem(title = "Payment", statusLabel = "In progress")
        )
    }

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(rememberScrollState())
        ) {
            PaymentBrandHeader(colors = colors, onMenuClick = onMenuClick)

            Ux4gDivider(color = colors.border)

            // Horizontal Progress Stepper
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = Ux4gSpace.Space16, vertical = Ux4gSpace.Space10)
            ) {
                Ux4gStepper(
                    totalSteps = 4,
                    currentStep = 4,
                    steps = steps,
                    orientation = StepperOrientation.Horizontal,
                    linePlacement = StepperLinePlacement.Center
                )
            }

            // Main Content Area
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = Ux4gSpace.Space16, vertical = Ux4gSpace.Space12)
            ) {
                Text(
                    text = "Choose payment method",
                    style = TextStyle(
                        fontSize = 22.sp,
                        fontWeight = FontWeight.ExtraBold,
                        color = colors.title,
                        letterSpacing = (-0.3).sp,
                        lineHeight = 28.sp
                    )
                )
                Spacer(Modifier.height(4.dp))
                Text(
                    text = "Select how you would like to pay $amountText.",
                    style = TextStyle(
                        fontSize = 13.sp,
                        color = colors.subtleText,
                        lineHeight = 18.sp
                    )
                )

                Spacer(Modifier.height(Ux4gSpace.Space20))

                // Payment Options List
                paymentOptions.forEach { option ->
                    val isSelected = option.id == selectedMethod
                    PaymentOptionTile(
                        item = option,
                        isSelected = isSelected,
                        colors = colors,
                        onSelect = { selectedMethod = option.id }
                    )
                    Spacer(Modifier.height(Ux4gSpace.Space8))
                }

                // Dynamic UPI ID Input Field
                if (selectedMethod == "upi") {
                    Spacer(Modifier.height(Ux4gSpace.Space4))
                    Ux4gInputField(
                        value = upiId,
                        onValueChange = { upiId = it },
                        label = "UPI ID",
                        placeholder = "username@bank / mobile@upi",
                        modifier = Modifier.fillMaxWidth()
                    )
                    Spacer(Modifier.height(Ux4gSpace.Space16))
                }

                // Security Assurance Banner powered by Ux4gStatusBanner
                Ux4gStatusBanner(
                    variant = Ux4gBannerVariant.InfoLight,
                    title = "Payment transactions are secured by 256-bit SSL encryption. We do not store your card details.",
                    leadingIcon = {
                        Icon(
                            imageVector = Icons.Default.Lock,
                            contentDescription = "Secured",
                            tint = colors.securityBannerIcon,
                            modifier = Modifier.size(18.dp)
                        )
                    },
                    backgroundColor = colors.securityBannerBg,
                    borderColor = colors.securityBannerBorder,
                    titleStyle = TextStyle(
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Medium,
                        color = colors.securityBannerText,
                        lineHeight = 16.sp
                    ),
                    margin = PaddingValues(0.dp),
                    modifier = Modifier.fillMaxWidth()
                )

                Spacer(Modifier.height(Ux4gSpace.Space20))

                // Action Buttons
                Ux4gButton(
                    text = "Pay $amountText",
                    onClick = { onPayClick(selectedMethod, upiId) },
                    variant = Ux4gButtonVariant.Primary,
                    size = Ux4gButtonSize.Large,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                )
                Spacer(Modifier.height(Ux4gSpace.Space12))

                Box(
                    modifier = Modifier.fillMaxWidth(),
                    contentAlignment = Alignment.Center
                ) {
                    Ux4gButton(
                        text = "Back",
                        onClick = onBackClick,
                        variant = Ux4gButtonVariant.Ghost,
                        size = Ux4gButtonSize.Medium,
                        modifier = Modifier.height(44.dp)
                    )
                }
            }

            Spacer(Modifier.weight(1f, fill = false))
            PaymentBrandFooter(colors = colors)
        }
    }
}

@Composable
private fun PaymentOptionTile(
    item: PaymentOptionItem,
    isSelected: Boolean,
    colors: PaymentColors,
    onSelect: () -> Unit
) {
    val bg = if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg
    val titleColor = if (isSelected) colors.optionSelectedTitle else colors.optionUnselectedTitle
    val subColor = if (isSelected) colors.optionSelectedSub else colors.optionUnselectedSub

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(bg)
            .clickable(onClick = onSelect)
            .padding(horizontal = 16.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Ux4gRadioButton(
            value = item.id,
            groupValue = if (isSelected) item.id else "",
            onValueChange = { onSelect() },
            size = Ux4gRadioButtonSize.Medium
        )
        Spacer(Modifier.width(12.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = item.title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = titleColor
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = item.subtitle,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}

@Composable
private fun PaymentOptionTileCard(
    item: PaymentOptionItem,
    isSelected: Boolean,
    colors: PaymentCardColors,
    onSelect: () -> Unit
) {
    val bg = if (isSelected) colors.optionSelectedBg else colors.optionUnselectedBg
    val titleColor = if (isSelected) colors.optionSelectedTitle else colors.optionUnselectedTitle
    val subColor = if (isSelected) colors.optionSelectedSub else colors.optionUnselectedSub

    Row(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(bg)
            .clickable(onClick = onSelect)
            .padding(horizontal = 16.dp, vertical = 14.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {
        Ux4gRadioButton(
            value = item.id,
            groupValue = if (isSelected) item.id else "",
            onValueChange = { onSelect() },
            size = Ux4gRadioButtonSize.Medium
        )
        Spacer(Modifier.width(12.dp))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = item.title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = titleColor
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = item.subtitle,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}
