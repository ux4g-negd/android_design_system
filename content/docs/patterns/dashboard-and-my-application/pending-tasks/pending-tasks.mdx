# Pending Tasks

A task-oriented pattern prioritizing actionable citizen interventions required to unblock government applications, renewals, and departmental reviews.

The screen highlights urgent citizen action items through clear status filter chips (`Ux4gChoiceChip`), urgency countdown pills (*Overdue by 2 days*, *Due in 5 days*, *Due in 9 days*), structured application reference metadata, and prominent execution buttons (`Ux4gButton`). A dedicated collapsible section records recently concluded actions (*Completed today*) with receipt access, ensuring transparent end-to-end petition progress.

/**
 * Task Card displaying actionable item details, urgency badge, and execution button.
 */
@Composable
private fun TaskCard(
    icon: ImageVector,
    title: String,
    subtitle: String,
    urgencyText: String,
    urgencyColor: Color,
    urgencyBg: Color,
    isExpanded: Boolean,
    onToggleExpand: () -> Unit,
    deadline: String? = null,
    reference: String? = null,
    department: String? = null,
    supportText: String? = null,
    onSupportClick: () -> Unit = {},
    actionText: String,
    onActionClick: () -> Unit,
    actionFilled: Boolean = true,
    cardBg: Color,
    borderCol: Color,
    textPrimary: Color,
    textSecondary: Color,
    primaryColor: Color
) {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(12.dp))
            .padding(16.dp)
    ) {
        Column(modifier = Modifier.fillMaxWidth()) {
            // Header Row
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .clickable { onToggleExpand() },
                verticalAlignment = Alignment.Top
            ) {
                Box(
                    modifier = Modifier
                        .size(32.dp)
                        .clip(RoundedCornerShape(4.dp))
                        .background(Color(0xFFF3F4F6)),
                    contentAlignment = Alignment.Center
                ) {
                    Icon(
                        imageVector = icon,
                        contentDescription = null,
                        tint = textSecondary,
                        modifier = Modifier.size(16.dp)
                    )
                }

                Spacer(modifier = Modifier.width(8.dp))

                Column(modifier = Modifier.weight(1f)) {
                    Text(
                        text = title,
                        style = TextStyle(
                            fontSize = 14.sp,
                            fontWeight = FontWeight.Bold,
                            color = textPrimary
                        )
                    )
                    Spacer(modifier = Modifier.height(2.dp))
                    Text(
                        text = subtitle,
                        style = TextStyle(
                            fontSize = 12.sp,
                            color = textSecondary
                        )
                    )
                }

                Icon(
                    imageVector = if (isExpanded) Icons.Default.KeyboardArrowUp else Icons.Default.KeyboardArrowDown,
                    contentDescription = if (isExpanded) "Collapse" else "Expand",
                    tint = textSecondary,
                    modifier = Modifier.size(16.dp)
                )
            }

            Spacer(modifier = Modifier.height(10.dp))

            // Urgency Badge
            Box(
                modifier = Modifier
                    .clip(RoundedCornerShape(4.dp))
                    .background(urgencyBg)
                    .padding(horizontal = 8.dp, vertical = 3.dp)
            ) {
                Text(
                    text = urgencyText,
                    style = TextStyle(
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Medium,
                        color = urgencyColor
                    )
                )
            }

            if (isExpanded && deadline != null) {
                Spacer(modifier = Modifier.height(10.dp))
                Divider(color = borderCol, thickness = 1.dp)
                Spacer(modifier = Modifier.height(12.dp))

                // Metadata Grid
                Row(modifier = Modifier.fillMaxWidth()) {
                    Column(modifier = Modifier.weight(1f)) {
                        Text(
                            text = "Deadline",
                            style = TextStyle(fontSize = 11.sp, color = textSecondary)
                        )
                        Spacer(modifier = Modifier.height(2.dp))
                        Text(
                            text = deadline,
                            style = TextStyle(
                                fontSize = 13.sp,
                                fontWeight = FontWeight.Medium,
                                color = textPrimary
                            )
                        )
                    }
                    Column(modifier = Modifier.weight(1f)) {
                        Text(
                            text = "Reference",
                            style = TextStyle(fontSize = 11.sp, color = textSecondary)
                        )
                        Spacer(modifier = Modifier.height(2.dp))
                        Text(
                            text = reference.orEmpty(),
                            style = TextStyle(
                                fontSize = 13.sp,
                                fontWeight = FontWeight.Medium,
                                color = textPrimary
                            )
                        )
                    }
                }

                Spacer(modifier = Modifier.height(12.dp))

                Row(modifier = Modifier.fillMaxWidth()) {
                    Column(modifier = Modifier.weight(1f)) {
                        Text(
                            text = "Department",
                            style = TextStyle(fontSize = 11.sp, color = textSecondary)
                        )
                        Spacer(modifier = Modifier.height(2.dp))
                        Text(
                            text = department.orEmpty(),
                            style = TextStyle(
                                fontSize = 13.sp,
                                fontWeight = FontWeight.Medium,
                                color = textPrimary
                            )
                        )
                    }
                    Column(modifier = Modifier.weight(1f)) {
                        Text(
                            text = "Have issues?",
                            style = TextStyle(fontSize = 11.sp, color = textSecondary)
                        )
                        Spacer(modifier = Modifier.height(2.dp))
                        Row(
                            verticalAlignment = Alignment.CenterVertically,
                            modifier = Modifier.clickable { onSupportClick() }
                        ) {
                            Text(
                                text = supportText.orEmpty(),
                                style = TextStyle(
                                    fontSize = 13.sp,
                                    fontWeight = FontWeight.Medium,
                                    color = primaryColor
                                )
                            )
                            Spacer(modifier = Modifier.width(4.dp))
                            Icon(
                                imageVector = Icons.Default.ArrowForward,
                                contentDescription = null,
                                tint = primaryColor,
                                modifier = Modifier.size(14.dp)
                            )
                        }
                    }
                }
            }

            Spacer(modifier = Modifier.height(16.dp))

            // Action Button
            Ux4gButton(
                text = actionText,
                onClick = onActionClick,
                variant = if (actionFilled) Ux4gButtonVariant.Primary else Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Default,
                modifier = Modifier.fillMaxWidth()
            )
        }
    }
}
