# Search Applications

A real-time search and query discovery pattern enabling citizens to rapidly locate specific petitions among multiple ongoing or completed government applications.

The screen features a citizen greeting and application metric summary cards (*Active*, *Needs attention*, *Completed*, *Total*), segmented filter chips (`Ux4gChoiceChip`), an interactive search bar with query clear action, matching result counter (*"2 results for 'income'"*), and expandable application result cards (`Ux4gResultRow`) highlighting urgent review deadlines (*8 days left*, *Under review*) and itemized petition metadata.

/**
 * Metric summary card displaying counts and status indicator dots.
 */
@Composable
private fun StatCard(
    value: String,
    label: String,
    modifier: Modifier = Modifier,
    dotColor: Color? = null,
    cardBg: Color,
    borderCol: Color,
    textColor: Color,
    subColor: Color
) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(10.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(10.dp))
            .padding(14.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.Start) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = value,
                    style = TextStyle(
                        fontSize = 24.sp,
                        fontWeight = FontWeight.Bold,
                        color = textColor
                    )
                )
                if (dotColor != null) {
                    Spacer(modifier = Modifier.width(6.dp))
                    Box(
                        modifier = Modifier
                            .size(8.dp)
                            .clip(CircleShape)
                            .background(dotColor)
                    )
                }
            }
            Spacer(modifier = Modifier.height(4.dp))
            Text(
                text = label,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}
