# My Applications

A centralized citizen dashboard pattern providing real-time visibility and status tracking across all submitted, in-review, and completed public service petitions.

The screen opens with a personalized citizen greeting and summary statistic counters (*Active*, *Needs attention*, *Completed*, *Total*). Filter chips with quantitative count badges (`Ux4gChoiceChip`) allow rapid segmentation (*All*, *Pending*, *Under Review*, *Approved*, *Rejected*), supported by an omnichannel search bar with voice recognition. Individual application entries are surfaced via expandable result cards (`Ux4gResultRow`) displaying critical metadata, urgency alert pills (*2 days overdue*, *Escalation available*, *Action needed*), and direct lifecycle actions (*Track*, *Download*).

/**
 * Metric summary card displaying status counts and alert indicators.
 */
@Composable
private fun StatCard(
    value: String,
    label: String,
    modifier: Modifier = Modifier,
    dotColor: Color? = null,
    cardBg: Color,
    borderCol: Color,
    textColor: Color,
    subColor: Color
) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(10.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(10.dp))
            .padding(14.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.Start) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = value,
                    style = TextStyle(
                        fontSize = 24.sp,
                        fontWeight = FontWeight.Bold,
                        color = textColor
                    )
                )
                if (dotColor != null) {
                    Spacer(modifier = Modifier.width(6.dp))
                    Box(
                        modifier = Modifier
                            .size(8.dp)
                            .clip(CircleShape)
                            .background(dotColor)
                    )
                }
            }
            Spacer(modifier = Modifier.height(4.dp))
            Text(
                text = label,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}
