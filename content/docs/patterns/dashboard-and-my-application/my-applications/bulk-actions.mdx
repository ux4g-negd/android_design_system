# Bulk Actions

A high-efficiency batch processing pattern designed for citizens and business representatives managing multiple ongoing or finalized government petitions.

The screen equips users with granular checkbox selection controls across application cards, anchored by a sticky or prominent bulk action toolbar. The banner displays the dynamic selection tally (*"3 selected"*), a quick filter reset trigger (*"Clear filters"*), and batch execution triggers—enabling citizens to perform simultaneous actions such as **Download all** approved documents or **Track together** across active department pipelines.

/**
 * Selectable application card with checkbox toggle and action controls.
 */
@Composable
private fun SelectableApplicationCard(
    index: Int,
    title: String,
    statusText: String,
    badgeText: String,
    badgeColor: Color,
    dotColor: Color,
    actionText: String,
    actionIcon: androidx.compose.ui.graphics.vector.ImageVector? = null,
    isSelected: Boolean,
    onToggle: (Boolean) -> Unit,
    cardBg: Color,
    borderCol: Color,
    textPrimary: Color
) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 4.dp)
            .clickable { onToggle(!isSelected) },
        verticalAlignment = Alignment.CenterVertically
    ) {
        // Selection Checkbox
        Box(
            modifier = Modifier
                .size(20.dp)
                .clip(RoundedCornerShape(4.dp))
                .background(if (isSelected) Ux4gColors.Primary600 else Color.Transparent)
                .border(
                    width = 1.5.dp,
                    color = if (isSelected) Ux4gColors.Primary600 else Color(0xFFD1D5DB),
                    shape = RoundedCornerShape(4.dp)
                ),
            contentAlignment = Alignment.Center
        ) {
            if (isSelected) {
                Icon(
                    imageVector = Icons.Default.Check,
                    contentDescription = "Selected",
                    tint = Color.White,
                    modifier = Modifier.size(14.dp)
                )
            }
        }

        Spacer(modifier = Modifier.width(10.dp))

        // Application Card Details
        Box(
            modifier = Modifier
                .weight(1f)
                .background(cardBg)
                .border(width = 0.dp, color = Color.Transparent)
                .padding(vertical = 10.dp, horizontal = 4.dp)
        ) {
            Column(modifier = Modifier.fillMaxWidth()) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = title.replace("\\n", " "),
                        style = TextStyle(
                            fontSize = 15.sp,
                            fontWeight = FontWeight.SemiBold,
                            color = textPrimary
                        ),
                        modifier = Modifier.weight(1f)
                    )

                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Ux4gButton(
                            text = actionText,
                            variant = Ux4gButtonVariant.Outline,
                            size = Ux4gButtonSize.Small,
                            onClick = { /* Handle action */ },
                            modifier = Modifier.height(32.dp)
                        )
                        Spacer(modifier = Modifier.width(6.dp))
                        Icon(
                            imageVector = Icons.Default.KeyboardArrowDown,
                            contentDescription = "Expand details",
                            tint = Color(0xFF6B7280),
                            modifier = Modifier.size(20.dp)
                        )
                    }
                }

                Spacer(modifier = Modifier.height(6.dp))

                // Status Pill Tag
                Ux4gUnifiedPillTag(
                    segments = listOfNotNull(
                        Ux4gPillSegment(
                            text = statusText,
                            textColor = textPrimary,
                            leading = {
                                Box(
                                    modifier = Modifier
                                        .size(8.dp)
                                        .clip(CircleShape)
                                        .background(dotColor)
                                )
                            }
                        ),
                        if (badgeText.isNotEmpty()) {
                            Ux4gPillSegment(
                                text = badgeText,
                                textColor = badgeColor
                            )
                        } else null
                    ),
                    modifier = Modifier.padding(top = 2.dp)
                )

                Spacer(modifier = Modifier.height(8.dp))
                Divider(color = borderCol, thickness = 1.dp)
            }
        }
    }
}

/**
 * Metric summary card displaying status counts and alert indicators.
 */
@Composable
private fun StatCard(
    value: String,
    label: String,
    modifier: Modifier = Modifier,
    dotColor: Color? = null,
    cardBg: Color,
    borderCol: Color,
    textColor: Color,
    subColor: Color
) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(10.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(10.dp))
            .padding(14.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.Start) {
            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = value,
                    style = TextStyle(
                        fontSize = 24.sp,
                        fontWeight = FontWeight.Bold,
                        color = textColor
                    )
                )
                if (dotColor != null) {
                    Spacer(modifier = Modifier.width(6.dp))
                    Box(
                        modifier = Modifier
                            .size(8.dp)
                            .clip(CircleShape)
                            .background(dotColor)
                    )
                }
            }
            Spacer(modifier = Modifier.height(4.dp))
            Text(
                text = label,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = subColor
                )
            )
        }
    }
}
