# Consent Management

A citizen empowerment and ongoing data governance pattern that allows users to review previously granted consents, inspect itemized sharing scopes across verified credentials and profile attributes, check active validity durations, access historical audit trails, and exercise their legal right to revoke or withdraw consent at any time.

The pattern presents clear, color-coded status badges for every permission category:
- **Shared**: Actively authorized attribute or document.
- **Partly shared**: Group permission where a subset of credentials has been granted.
- **Not shared**: Explicitly withheld attribute or document.

@Composable
private fun ConsentStatusPill(status: ConsentShareStatus, isDark: Boolean) {
    val (bg, textColor) = when (status) {
        ConsentShareStatus.Shared -> {
            if (isDark) Color(0xFF14532D) to Color(0xFF86EFAC)
            else Color(0xFFDCFCE7) to Color(0xFF166534)
        }
        ConsentShareStatus.PartlyShared -> {
            if (isDark) Color(0xFF452205) to Color(0xFFFFB74D)
            else Color(0xFFFFE8C6) to Color(0xFF975A16)
        }
        ConsentShareStatus.NotShared -> {
            if (isDark) Ux4gColors.Neutral800 to Ux4gColors.Neutral300
            else Color(0xFFE9ECEF) to Color(0xFF495057)
        }
    }

    Box(
        modifier = Modifier
            .clip(RoundedCornerShape(6.dp))
            .background(bg)
            .padding(horizontal = 8.dp, vertical = 3.dp)
    ) {
        Text(
            text = status.label,
            style = TextStyle(
                fontSize = 11.sp,
                fontWeight = FontWeight.SemiBold,
                color = textColor
            )
        )
    }
}

@Composable
private fun ConsentStatusPillCard(status: ConsentShareStatus, isDark: Boolean) {
    val (bg, textColor) = when (status) {
        ConsentShareStatus.Shared -> {
            if (isDark) Color(0xFF14532D) to Color(0xFF86EFAC)
            else Color(0xFFDCFCE7) to Color(0xFF166534)
        }
        ConsentShareStatus.PartlyShared -> {
            if (isDark) Color(0xFF452205) to Color(0xFFFFB74D)
            else Color(0xFFFFE8C6) to Color(0xFF975A16)
        }
        ConsentShareStatus.NotShared -> {
            if (isDark) Ux4gColors.Neutral800 to Ux4gColors.Neutral300
            else Color(0xFFE9ECEF) to Color(0xFF495057)
        }
    }

    Box(
        modifier = Modifier
            .clip(RoundedCornerShape(6.dp))
            .background(bg)
            .padding(horizontal = 8.dp, vertical = 3.dp)
    ) {
        Text(
            text = status.label,
            style = TextStyle(
                fontSize = 11.sp,
                fontWeight = FontWeight.SemiBold,
                color = textColor
            )
        )
    }
}
