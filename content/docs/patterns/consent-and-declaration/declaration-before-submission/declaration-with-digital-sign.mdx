# Declaration with Digital Sign

A legally binding digital affirmation and signature pattern used for high-stakes citizen service applications, permit issuances, and official statutory filings.

This pattern combines solemn legal declarations with digital name attestation. Citizens must review the declaration statements, check the mandatory agreement confirmation, and type their full name exactly as registered in official records (such as Aadhaar). A real-time verification indicator validates the signature before enabling the final application submission trigger, reinforced with a statutory penal caution under **Section 193 of the Indian Penal Code (IPC)**.

    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Ux4gColors.Neutral900
    val textSecondary = if (isDark) Ux4gColors.Neutral400 else Ux4gColors.Neutral600

    val inputBorderColor = when {
        isSigned -> if (isDark) Ux4gColors.Green400 else Ux4gColors.Green600
        else -> if (isDark) Ux4gColors.Neutral600 else Ux4gColors.Neutral300
    }

    val warningBg = if (isDark) Color(0xFF3B1E08) else Color(0xFFFFF7ED)
    val warningBorder = if (isDark) Color(0xFF7C2D12) else Color(0xFFFED7AA)
    val warningText = if (isDark) Color(0xFFFDBA74) else Color(0xFF9A3412)

    val validationColor = if (isSigned) {
        if (isDark) Ux4gColors.Green400 else Ux4gColors.Green600
    } else {
        if (isDark) Ux4gColors.Neutral400 else Ux4gColors.Neutral500
    }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(surfaceBg)
    ) {
        // App Header
        Ux4gAppHeader(
            title = "Declaration & Attestation",
            variant = Ux4gAppHeaderVariant.Standard,
            navigationAction = Ux4gAppHeaderAction(
                icon = Icons.Filled.ArrowBack,
                contentDescription = "Back",
                onClick = onBackClick
            )
        )

        // Scrollable Content
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(scrollState)
                .padding(horizontal = 20.dp, vertical = 24.dp)
        ) {
            // Heading
            Text(
                text = "Declaration",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = textPrimary,
                    letterSpacing = (-0.5).sp
                )
            )

            Spacer(modifier = Modifier.height(10.dp))

            // Dynamic Subtitle
            Text(
                text = if (isSigned) {
                    "Declaration agreed and signed. Your application is ready to submit."
                } else {
                    "You have agreed. Now sign by typing your full name as per Aadhaar."
                },
                style = TextStyle(
                    fontSize = 14.sp,
                    color = textSecondary,
                    lineHeight = 20.sp,
                    fontWeight = FontWeight.Normal
                )
            )

                    text = "4. I authorise the Revenue Department to verify these particulars from any source.\n\n" +

                        fontWeight = FontWeight.Medium

            }

                    .fillMaxWidth()

                contentAlignment = Alignment.CenterStart
            ) {

                    modifier = Modifier.fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically

                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {

                    text = "Powered by Digital India",
                    style = TextStyle(
                        fontSize = 12.sp,
                        color = Ux4gColors.Neutral400,
                        fontWeight = FontWeight.Medium
                    )
                )
            }
