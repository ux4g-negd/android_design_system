# Inline Feedback Verification

A real-time status communication and timeline verification pattern designed to give citizens immediate transparency while backend automated systems inspect uploaded credentials (such as income certificates, domicile proofs, or disability records).

Instead of displaying an indeterminate, opaque loading spinner, this pattern breaks down document verification into granular, understandable checkpoints (file limits, image legibility, OCR figure matching). Citizens see a live progress bar alongside a timeline checklist with status tags (*"Passed"*, *"Verifying"*), keeping them informed and engaged while processing occurs.

The pattern is available in two structural configurations:
1. **Default Style**: Clean, full-screen layout anchored by the official government portal header, back navigation, progress header, active document summary card, live timeline checklist, and a primary tracking action button.
2. **Card Style**: An elevated card container situated on a soft tinted purple background (`Ux4gColors.primary100` / `0xFFF2EFFF` in light mode, `primary800` in dark mode) with pinned bottom progression controls.

@Composable
private fun VerificationTimelineRow(
    step: VerificationCheckStep,
    isLast: Boolean,
    colors: InlineFeedbackVerificationColors
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.Top
    ) {
        // Timeline Indicator Column
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
            modifier = Modifier.width(28.dp)
        ) {
            if (step.isCompleted) {
                Icon(
                    imageVector = Icons.Default.CheckCircle,
                    contentDescription = "Passed",
                    tint = Ux4gColors.green600,
                    modifier = Modifier.size(20.dp)
                )
            } else {
                Icon(
                    imageVector = Icons.Default.Schedule,
                    contentDescription = "Verifying",
                    tint = Ux4gColors.orange500,
                    modifier = Modifier.size(20.dp)
                )
            }
            if (!isLast) {
                Box(
                    modifier = Modifier
                        .width(2.dp)
                        .height(44.dp)
                        .background(colors.border)
                )
            }
        }

        Spacer(modifier = Modifier.width(10.dp))

        // Step Content & Status Badge
        Column(modifier = Modifier.weight(1f)) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.Top
            ) {
                Column(modifier = Modifier.weight(1f)) {
                    Text(
                        text = step.title,
                        style = TextStyle(
                            fontSize = 14.sp,
                            fontWeight = FontWeight.SemiBold,
                            color = colors.title
                        )
                    )
                    Spacer(modifier = Modifier.height(2.dp))
                    Text(
                        text = step.date,
                        style = TextStyle(
                            fontSize = 12.sp,
                            color = colors.subtleText
                        )
                    )
                }
                Spacer(modifier = Modifier.width(8.dp))
                if (step.isCompleted) {
                    Ux4gTag(
                        text = "Passed",
                        backgroundColor = colors.tagPassedBg,
                        contentColor = colors.tagPassedText
                    )
                } else {
                    Ux4gTag(
                        text = "Verifying",
                        backgroundColor = colors.tagVerifyingBg,
                        contentColor = colors.tagVerifyingText
                    )
                }
            }
        }
    }
}
