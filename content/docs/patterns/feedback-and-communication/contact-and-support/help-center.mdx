# Help Center

A foundational self-service support pattern that enables citizens to independently troubleshoot issues, locate answers to frequently asked questions, and navigate public welfare workflows.

The pattern pairs a high-visibility search bar (`Ux4gSearchField` with voice search and instant submit) with structured category navigation (*Application Issues*, *Documents and Uploads*, *Payments and Fees*, *Account and Profile*, *Technical Issues*). Citizens can quickly diagnose queries without waiting in physical administrative queues or submitting formal grievances.

The pattern is available in two structural configurations:
1. **Default Style**: Clean, direct layout situated on a neutral white canvas featuring the official government portal header, prominent search field, categorized topic card with dividers, and Digital India branding.
2. **Card Style**: An elevated card variant situated on a soft lavender canvas (`0xFFE2DCFF` / `Ux4gColors.primary100` in light mode, `primary800` in dark mode) with enclosed card container and footer attribution.

/**
 * Help Center Screen - Default Style
 *
 * @param onCategoryClick Invoked when a citizen taps a category row
 * @param onSearchSubmit Invoked when a citizen executes a search
 */
@Composable
fun HelpCenterScreen(
    categories: List<HelpCategory> = defaultHelpCategories,
    onCategoryClick: (HelpCategory) -> Unit = {},
    onSearchSubmit: (String) -> Unit = {}
) {
    val isDark = isSystemInDarkTheme()
    val colors = if (isDark) DarkHelpCenterColors else LightHelpCenterColors

    var searchQuery by remember { mutableStateOf("") }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        // App Header
        Column {
            Ux4gAppHeader(
                title = "",
                variant = Ux4gAppHeaderVariant.Light,
                height = 64.dp,
                horizontalPadding = Ux4gSpace.Space16
            )
            Ux4gDivider(
                color = colors.headerDivider,
                thickness = 1.dp
            )
        }

        // Scrollable Help Center Content
        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 20.dp)
        ) {
            // Heading & Subtitle
            Text(
                text = "Help Centre",
                style = TextStyle(
                    fontSize = 22.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.titleColor
                )
            )

            Spacer(modifier = Modifier.height(8.dp))

            Text(
                text = "Find answers to common questions or contact our support team.",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtitleColor,
                    lineHeight = 20.sp
                )
            )

            Spacer(modifier = Modifier.height(20.dp))

            // Search Field with Submit & Voice
            Ux4gSearchField(
                value = searchQuery,
                onValueChange = { searchQuery = it },
                placeholder = "Search for...",
                variant = Ux4gSearchFieldVariant.SearchWithSubmit,
                showVoiceIcon = true,
                showClearIcon = true,
                onClearClick = { searchQuery = "" },
                onSubmitClick = { onSearchSubmit(searchQuery) },
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(24.dp))

            // Section: Browse by category
            Text(
                text = "Browse by category",
                style = TextStyle(
                    fontSize = 15.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.sectionHeaderColor
                )
            )

            Spacer(modifier = Modifier.height(12.dp))

            // Category List Card Container
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .shadow(elevation = 3.dp, shape = RoundedCornerShape(10.dp))
                    .clip(RoundedCornerShape(10.dp))
                    .background(colors.cardBackground)
                    .border(1.dp, colors.cardBorder, RoundedCornerShape(10.dp))

                Column {
                    categories.forEachIndexed { index, category ->
                        Row(
                            modifier = Modifier
                                .fillMaxWidth()
                                .clickable { onCategoryClick(category) }
                                .padding(horizontal = 16.dp, vertical = 14.dp),
                            horizontalArrangement = Arrangement.SpaceBetween,
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Text(
                                text = category.title,
                                style = TextStyle(
                                    fontSize = 14.sp,
                                    fontWeight = FontWeight.SemiBold,
                                    color = colors.itemTextColor
                                ),
                                modifier = Modifier.weight(1f)
                            )
                            Icon(
                                imageVector = Icons.AutoMirrored.Filled.KeyboardArrowRight,
                                contentDescription = "Navigate to \${category.title}",
