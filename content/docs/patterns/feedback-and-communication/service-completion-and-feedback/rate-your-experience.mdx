# Rate your Experience

A post-service citizen feedback pattern that prompts citizens to evaluate their digital experience immediately following the completion of a public service workflow (such as document issuance, license renewal, or appointment booking).

This pattern integrates the official **`Ux4gFeedbackFormStar`** component, featuring an interactive 5-star scoring scale with dynamic sentiment color thresholds (accent red for critical ratings 1–2, warm yellow for positive ratings 3–5), selectable improvement category chips (*"Content accuracy"*, *"Visual design"*, *"Performance"*, *"Navigation"*), an optional multiline comment field, and responsive submit and skip actions.

The pattern is available in two structural configurations:
1. **Default Style**: Clean, direct layout situated on a pure neutral background with `Ux4gAppHeader`, embedded `Ux4gFeedbackFormStar`, and Digital India branding.
2. **Card Style**: An elevated presentation placed on a soft lavender background (`Ux4gColors.primary100` / `0xFFF2EFFF` in light mode, `primary800` in dark mode) with centered feedback form and footer attribution.

val defaultImprovementOptions = listOf(
    "Content accuracy",
    "Visual design",
    "Performance",
    "Navigation"
)

/**
 * Rate your Experience Screen - Default Style
 *
 * Utilizes the Ux4gFeedbackFormStar component from the UX4G Design System.
 *
 * @param onSubmit Invoked when citizen submits rating, selected tags, and comments
 * @param onSkip Invoked when citizen skips feedback
 */
@Composable
fun RateExperienceScreen(
    onSubmit: (rating: Int, tags: List<String>, comment: String) -> Unit = { _, _, _ -> },
    onSkip: () -> Unit = {}
) {
    val isDark = isSystemInDarkTheme()
    val screenBackground = if (isDark) Ux4gColors.neutral900 else Ux4gColors.neutral0
    val headerDividerColor = if (isDark) Ux4gColors.neutral700 else Ux4gColors.neutral300

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(screenBackground)
    ) {
        // App Header
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16
        )
        Ux4gDivider(
            color = headerDividerColor,
            thickness = 1.dp
        )

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 20.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            // Star Feedback Form Component
            Ux4gFeedbackFormStar(
                modifier = Modifier.fillMaxWidth(),
                title = "Rate your experience",
                improvementTitle = "What can we improve?",
                improvementOptions = defaultImprovementOptions,
                commentPlaceholder = "Tell us more about your experience",
                submitButtonText = "Submit",
                skipButtonText = "Skip",
                onSubmit = onSubmit,
                onSkip = onSkip
            )
        }

        // Powered by Digital India Footer
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 16.dp),
            horizontalArrangement = Arrangement.Center,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = "Powered by - Digital India",
                style = TextStyle(
                    fontSize = 11.sp,
                    color = Color(0xFF6B7280),
                    fontWeight = FontWeight.Medium
                )
            )
        }
    }
}

}
