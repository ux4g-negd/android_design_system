# Rate Experience Card

A comprehensive feedback collection card pattern that enables public service agencies to capture nuanced citizen sentiment following service completion.

This pattern integrates the official **`Ux4gFeedbackFormStar`** component suite. In addition to quantitative 5-star rating with dynamic sentiment color thresholds, it offers multi-select improvement chips (*"Content accuracy"*, *"Visual design"*, *"Performance"*, *"Navigation"*) and an optional multiline comment field (*"Please provide your valuable feedback on how we can improve our portal for you."*). This provides actionable diagnostics to government product teams while allowing citizens who prefer brevity to submit or skip with zero friction.

The pattern is available in two structural configurations:
1. **Default Style**: Clean, direct layout on a white canvas featuring the official government portal header, contained `Ux4gFeedbackFormStar` card, and Digital India branding.
2. **Card Style**: An elevated card variant situated on a soft lavender background (`Ux4gColors.primary100` / `0xFFF2EFFF` in light mode, `primary800` in dark mode) with centered footer attribution.

val defaultImprovementOptions = listOf(
    "Content accuracy",
    "Visual design",
    "Performance",
    "Navigation"
)

/**
 * Rate Experience Card Screen - Default Style
 *
 * Utilizes the official Ux4gFeedbackFormStar component suite from the UX4G Design System.
 *
 * @param onSubmit Invoked when citizen submits rating, selected tags, and comments
 * @param onSkip Invoked when citizen skips feedback
 */
@Composable
fun FeedbackRatingScreen(
    onSubmit: (rating: Int, tags: List<String>, comment: String) -> Unit = { _, _, _ -> },
    onSkip: () -> Unit = {}
) {
    val isDark = isSystemInDarkTheme()
    val screenBackground = if (isDark) Ux4gColors.neutral900 else Ux4gColors.neutral0
    val headerDividerColor = if (isDark) Ux4gColors.neutral700 else Ux4gColors.neutral300

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(screenBackground)
    ) {
        // App Header
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16
        )
        Ux4gDivider(
            color = headerDividerColor,
            thickness = 1.dp
        )

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 20.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            // Star Feedback Form Component
            Ux4gFeedbackFormStar(
                modifier = Modifier.fillMaxWidth(),
                title = "Rate your experience",
                improvementTitle = "What can we improve?",
                improvementOptions = defaultImprovementOptions,
                commentPlaceholder = "Please provide your valuable feedback on how we can improve our portal for you.",
                submitButtonText = "Submit",
                skipButtonText = "Skip",
                onSubmit = onSubmit,
                onSkip = onSkip
            )
        }

        // Powered by Digital India Footer
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(bottom = 16.dp),
            horizontalArrangement = Arrangement.Center,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = "Powered by - Digital India",
                style = TextStyle(
                    fontSize = 11.sp,
                    color = Color(0xFF6B7280),
                    fontWeight = FontWeight.Medium
                )
            )
        }
    }
}

}
