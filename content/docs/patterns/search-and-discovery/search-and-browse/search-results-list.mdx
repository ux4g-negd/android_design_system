# Search Results List

A structured results listing pattern designed for efficient evaluation and filtering of high-volume public services, welfare schemes, and civil certificates.

The screen pairs an active keyword search bar (`Ux4gSearchField`) with quantitative feedback (*"48 results"*) and a dedicated **Filters** modal trigger (`Ux4gButton` with `Icons.Default.Tune`). Horizontally scrollable segmentation chips (`Ux4gChoiceChip`) allow rapid pivoting between categories (*All*, *Certificates*, *Schemes*, *Services*, *Documents*, *Applications*). Individual service cards display immediate cost indicators (*Free* or *Paid* status badges), statutory turnaround times, and direct **Apply** actions, concluded by accessible pagination controls.

/**
 * Search Results List Screen Pattern
 *
 * @param onBack Invoked when citizen taps back button
 * @param onSearchSubmit Invoked when citizen submits search
 * @param onOpenFilters Invoked when citizen taps 'Filters' button
 * @param onApplyService Invoked when citizen taps 'Apply' on a service card
 * @param onPageChange Invoked when citizen changes pagination page
 */
@Composable
fun SearchResultsListScreen(
    onBack: () -> Unit = {},
    onSearchSubmit: (String) -> Unit = {},
    onOpenFilters: () -> Unit = {},
    onApplyService: (String) -> Unit = {},
    onPageChange: (Int) -> Unit = {}
) {

    val chipScrollState = rememberScrollState()

    var selectedChipIndex by remember { mutableIntStateOf(1) } // "Certificates · 22" selected
    var currentPage by remember { mutableIntStateOf(1) }

    val surfaceBg = if (isDark) Ux4gColors.Neutral950 else Ux4gColors.Neutral50
    val cardBg = if (isDark) Ux4gColors.Neutral900 else Color.White
    val borderCol = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)
    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Ux4gColors.Neutral900
    val textSecondary = if (isDark) Ux4gColors.Neutral400 else Color(0xFF4B5563)
    val badgeBg = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)

    val filterChips = listOf(
        "All · 48",
        "Certificates · 22",
        "Schemes · 14",
        "Services · 12",
        "Documents · 8",
        "Applications · 5"
    )

    val results = listOf(
        SearchResultItem("Income Certificate", "Free", null, "20 mins", false),
        SearchResultItem("Income Tax Return (ITR) Filing", "Free", null, "30 mins", false),
        SearchResultItem("Low Income Group (LIG) Certificate", "Free", null, "7 days", false),
        SearchResultItem("Income & Asset Certificate", "Paid", "₹50+", "15 days", true),
        SearchResultItem("EWS Certificate", "Free", null, "10 days", false)
    )

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(surfaceBg)
    ) {
        // App Header - Filled Primary Style
        Ux4gAppHeader(
            variant = Ux4gAppHeaderVariant.Filled,
            title = "National Services Portal",
            showBackButton = true,
            onBackClick = onBack
        )

        // Search Bar Section
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 12.dp)
        ) {
            Ux4gSearchField(
                value = searchQuery,
                onValueChange = { searchQuery = it },
                placeholder = "Search services...",
                variant = Ux4gSearchFieldVariant.SearchWithSubmit,
                size = Ux4gSearchFieldSize.Medium,
                showVoiceIcon = true,
                showClearIcon = true,
                onClearClick = { searchQuery = "" },
                onSubmitClick = { onSearchSubmit(searchQuery) },
                modifier = Modifier.fillMaxWidth()
            )
        }

        // Results Header Row (Result count + Filter button)
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 4.dp),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text(
                text = "48 results",
                style = TextStyle(
                    fontSize = 15.sp,
                    fontWeight = FontWeight.Bold,
                    color = textPrimary
                )
            )

            Ux4gButton(
                text = "Filters",
                onClick = onOpenFilters,
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Small,
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Default.Tune,
                        contentDescription = "Filters",
                        tint = textPrimary,
                        modifier = Modifier.size(16.dp)
                    )
                }
            )
        }

        Spacer(modifier = Modifier.height(8.dp))

        // Horizontal Category Filter Chips
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .horizontalScroll(chipScrollState)
                .padding(horizontal = 16.dp),
            horizontalArrangement = Arrangement.spacedBy(8.dp)
        ) {
            filterChips.forEachIndexed { index, label ->
                Ux4gChoiceChip(
                    text = label,
                    selected = selectedChipIndex == index,
                    onClick = { selectedChipIndex = index },
                    size = Ux4gChoiceChipSize.Default
                )
            }
        }

        Spacer(modifier = Modifier.height(12.dp))

        // Paginated Service Results List
        Column(
            modifier = Modifier
                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(scrollState)
                .padding(horizontal = 16.dp),
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            results.forEach { item ->
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(RoundedCornerShape(12.dp))
                        .background(cardBg)
                        .border(1.dp, borderCol, RoundedCornerShape(12.dp))
                        .padding(14.dp)
                ) {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Column(modifier = Modifier.weight(1f)) {
                            Text(
                                text = item.title,
                                style = TextStyle(
                                    fontSize = 14.sp,
                                    fontWeight = FontWeight.Bold,
                                    color = textPrimary
                                )
                            )

                            Spacer(modifier = Modifier.height(6.dp))

                            Row(verticalAlignment = Alignment.CenterVertically) {
                                // Badge: Free / Paid
                                Box(
                                    modifier = Modifier
                                        .clip(RoundedCornerShape(4.dp))
                                        .background(badgeBg)
                                        .padding(horizontal = 6.dp, vertical = 2.dp)
                                ) {
                                    Text(
                                        text = item.statusTag,
                                        style = TextStyle(
                                            fontSize = 11.sp,
                                            fontWeight = FontWeight.SemiBold,
                                            color = textPrimary
                                        )
                                    )
                                }

                                if (item.isPaid && item.feeText != null) {
                                    Spacer(modifier = Modifier.width(4.dp))
                                    Text(
                                        text = item.feeText,
                                        style = TextStyle(
                                            fontSize = 12.sp,
                                            color = textSecondary
                                        )
                                    )
                                }

                                Spacer(modifier = Modifier.width(8.dp))

                                Icon(
                                    imageVector = Icons.Default.AccessTime,
                                    contentDescription = "Processing Time",
                                    tint = textSecondary,
                                    modifier = Modifier.size(13.dp)
                                )

                                Spacer(modifier = Modifier.width(3.dp))

                                Text(
                                    text = item.time,
                                    style = TextStyle(
                                        fontSize = 12.sp,
                                        color = textSecondary
                                    )
                                )
                            }
                        }

                        Spacer(modifier = Modifier.width(12.dp))

                        Ux4gButton(
                            text = "Apply",
                            onClick = { onApplyService(item.title) },
                            variant = Ux4gButtonVariant.Outline,
                            size = Ux4gButtonSize.Small

                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Small,
                enabled = currentPage > 1
            )

            Spacer(modifier = Modifier.width(16.dp))

            Text(
                text = "Page " + currentPage + " of 8",
                style = TextStyle(
                    fontSize = 13.sp,
                    color = textSecondary
                )
            )

            Spacer(modifier = Modifier.width(16.dp))

            Ux4gButton(
                text = "Next",
                onClick = {
                    if (currentPage < 8) {
                        currentPage++
                        onPageChange(currentPage)
                    }
                },
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Small,
                enabled = currentPage < 8
            )
        }
    }
}

private data class SearchResultItem(

    val statusTag: String,
    val feeText: String?,
    val time: String,
    val isPaid: Boolean
