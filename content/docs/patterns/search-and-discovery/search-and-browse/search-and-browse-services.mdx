# Search and Browse Services

A discovery-first landing pattern guiding citizens to relevant public services, subsidies, and certifications across departmental catalogs.

The screen anchors discovery with an omnichannel search bar (`Ux4gInputField` with keyword search and voice input), followed by a 2-column thematic category grid (*Health*, *Agriculture*, *Education*, *Land Records*). Prominent discovery sections highlight timely national interventions (*Featured Scheme* with direct eligibility checking), high-volume public petitions (*Popular Services* with fee, processing SLA, and quick apply actions), and an intuitive guided assessment callout (*Find my schemes* quiz) to assist citizens uncertain of their entitlements.

/**

    val featuredBorder = if (isDark) Color(0xFF432CBB) else Color(0xFFDDD6FE)
    val quizBg = if (isDark) Color(0xFF1E1435) else Color(0xFFF5F3FF)
    val quizBorder = if (isDark) Color(0xFF5B3FE8) else Color(0xFFC4B5FD)
    val greenColor = if (isDark) Color(0xFF22C55E) else Ux4gColors.Green600

    val categories = listOf(
        CategoryItem("Health", "24 services", Icons.Default.HealthAndSafety),
        CategoryItem("Agriculture", "18 services", Icons.Default.Agriculture),
        CategoryItem("Education", "31 services", Icons.Default.School),

    )

    val popularServices = listOf(
        PopularServiceItem("Income Certificate", "Revenue Department", "Free", "20 mins"),
        PopularServiceItem("Caste Certificate", "Social Welfare Dept", "Free", "15 mins"),

    Column(

    ) {

        Ux4gAppHeader(
            variant = Ux4gAppHeaderVariant.Filled,

            showBackButton = true,

        )

        Column(

                .weight(1f)
                .fillMaxWidth()
                .verticalScroll(scrollState)
                .padding(16.dp)
        ) {

                text = "Find any government service",
                style = TextStyle(
                    fontSize = 20.sp,
                    fontWeight = FontWeight.ExtraBold,
                    color = textPrimary
                )
            )

            Spacer(modifier = Modifier.height(4.dp))

            Text(
                text = "Search 3,000+ services & schemes",
                style = TextStyle(
                    fontSize = 13.sp,
                    color = textSecondary
                )
            )

            Spacer(modifier = Modifier.height(16.dp))

            Ux4gInputField(
                value = searchQuery,
                onValueChange = { searchQuery = it },
                placeholder = "Search services...",
                leadingIcon = {
                    Icon(
                        imageVector = Icons.Default.Search,
                        contentDescription = "Search",
                        tint = textSecondary,
                        modifier = Modifier.size(20.dp)
                    )
                },
                trailingIcon = {
                    Icon(
                        imageVector = Icons.Default.Mic,
                        contentDescription = "Voice Search",
                        tint = textSecondary,
                        modifier = Modifier.size(20.dp)
                    )
                },
                modifier = Modifier.fillMaxWidth()
            )

            Spacer(modifier = Modifier.height(24.dp))

            // Browse by category
            Text(
                text = "Browse by category",
                style = TextStyle(
                    fontSize = 16.sp,
                    fontWeight = FontWeight.Bold,
                    color = textPrimary
                )
            )

            Spacer(modifier = Modifier.height(12.dp))

            // 2-Column Category Grid
            Column(verticalArrangement = Arrangement.spacedBy(12.dp)) {
                for (i in categories.indices step 2) {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.spacedBy(12.dp)
                    ) {
                        CategoryCard(

                            onClick = { onSelectCategory(categories[i].label) },
                            cardBg = cardBg,
                            borderCol = borderCol,
                            primaryColor = primaryColor,
                            textPrimary = textPrimary,
                            textSecondary = textSecondary,
                            modifier = Modifier.weight(1f)
                        )
                        if (i + 1 < categories.size) {
                            CategoryCard(
                                item = categories[i + 1],
                                onClick = { onSelectCategory(categories[i + 1].label) },
                                cardBg = cardBg,
                                borderCol = borderCol,
                                primaryColor = primaryColor,
                                textPrimary = textPrimary,
                                textSecondary = textSecondary,
                                modifier = Modifier.weight(1f)

                    )
                }
            }
        }
    }

private data class CategoryItem(val label: String, val count: String, val icon: ImageVector)
private data class PopularServiceItem(val title: String, val department: String, val fee: String, val duration: String)

@Composable
private fun CategoryCard(
    item: CategoryItem,
    onClick: () -> Unit,
    cardBg: Color,
    borderCol: Color,
    primaryColor: Color,
    textPrimary: Color,
    textSecondary: Color,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(12.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(12.dp))
            .clickable { onClick() }
            .padding(14.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.Start) {
            Icon(
                imageVector = item.icon,
                contentDescription = item.label,
                tint = primaryColor,
                modifier = Modifier.size(24.dp)
            )
            Spacer(modifier = Modifier.height(8.dp))
            Text(
                text = item.label,
                style = TextStyle(
                    fontSize = 13.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = textPrimary
                )
            )
            Spacer(modifier = Modifier.height(2.dp))
            Text(
                text = item.count,
                style = TextStyle(
                    fontSize = 11.sp,
                    color = textSecondary
                )
            )
        }
    }
}
