# Search with Results

An intelligent search-as-you-type pattern accelerating public service discovery through instant autocomplete suggestions and departmental routing.

As citizens input partial terms into the omnichannel search bar (`Ux4gSearchField`), the interface dynamically projects a matched suggestions card. Each suggestion item highlights the primary petition title alongside its managing ministry or departmental authority (*Revenue Dept*, *Income Tax Dept*, *Social Welfare*). Below the search header, the page maintains situational context with an expanded category directory (*Health*, *Agriculture*, *Education*, *Transport*, *Finance*) and featured high-demand service cards.

 * @param onApplyService Invoked when citizen taps 'Apply' on a service card

    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Ux4gColors.Neutral900
    val textSecondary = if (isDark) Ux4gColors.Neutral400 else Color(0xFF4B5563)
    val textMuted = if (isDark) Ux4gColors.Neutral500 else Color(0xFF6B7280)
    val primaryColor = if (isDark) Ux4gColors.Primary300 else Ux4gColors.Primary600
    val greenColor = if (isDark) Color(0xFF22C55E) else Ux4gColors.Green600

    val suggestions = listOf(

        SuggestionItem("Income Tax Return Filing", "Income Tax Dept"),
        SuggestionItem("Low Income Group Certificate", "Social Welfare"),
        SuggestionItem("Income & Asset Certificate", "Revenue Dept")
    )

    val categories = listOf(
        CategoryItem("Health", "24 services", Icons.Default.HealthAndSafety),
        CategoryItem("Agriculture", "18 services", Icons.Default.Agriculture),
        CategoryItem("Education", "31 services", Icons.Default.School),
        CategoryItem("Land Records", "12 services", Icons.Default.Landscape),
        CategoryItem("Transport", "15 services", Icons.Default.DirectionsBus),
        CategoryItem("Social Welfare", "27 services", Icons.Default.People),
        CategoryItem("Utilities", "9 services", Icons.Default.Bolt),
        CategoryItem("Finance", "21 services", Icons.Default.AccountBalance)
    )

    val popularServices = listOf(
        PopularServiceItem("Income Certificate", "Revenue Department", "Free", "20 mins"),
        PopularServiceItem("Caste Certificate", "Social Welfare Dept", "Free", "15 mins"),
        PopularServiceItem("Driving Licence", "Transport Department", "₹200", "30 days")
    )

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(surfaceBg)
    ) {
        // App Header - Filled Primary Style
        Ux4gAppHeader(
            variant = Ux4gAppHeaderVariant.Filled,
            title = "National Services Portal",
            showBackButton = true,
            onBackClick = onBack
        )

            modifier = Modifier

                .fillMaxWidth()
                .verticalScroll(scrollState)
        ) {
            // Search Area with Tinted Background
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .background(headerSearchBg)
                    .padding(16.dp)
            ) {
                Text(
                    text = "Find any government service",
                    style = TextStyle(
                        fontSize = 18.sp,
                        fontWeight = FontWeight.ExtraBold,
                        color = textPrimary
                    )
                )

                Spacer(modifier = Modifier.height(4.dp))

                Text(
                    text = "Search 3,000+ services & schemes",
                    style = TextStyle(
                        fontSize = 13.sp,
                        color = textSecondary
                    )
                )

                Spacer(modifier = Modifier.height(16.dp))

                // Search field with submit button
                Ux4gSearchField(
                    value = searchValue,
                    onValueChange = { searchValue = it },
                    placeholder = "Search services...",
                    variant = Ux4gSearchFieldVariant.SearchWithSubmit,
                    size = Ux4gSearchFieldSize.Medium,
                    showVoiceIcon = true,
                    showClearIcon = true,
                    onClearClick = { searchValue = "" },
                    onSubmitClick = { onSearchSubmit(searchValue) },
                    modifier = Modifier.fillMaxWidth()
                )

                Spacer(modifier = Modifier.height(8.dp))

                // Autocomplete Suggestions Panel
                Box(
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(RoundedCornerShape(8.dp))
                        .background(cardBg)
                        .border(1.dp, borderCol, RoundedCornerShape(8.dp))
                        .padding(horizontal = 12.dp, vertical = 10.dp)
                ) {
                    Column(modifier = Modifier.fillMaxWidth()) {
                        // Searching query indicator
                        Text(
                            text = buildAnnotatedString {
                                append("Searching for ")
                                withStyle(SpanStyle(fontWeight = FontWeight.Bold)) {
                                    append("'$searchValue'")
                                }
                            },
                            style = TextStyle(
                                fontSize = 12.sp,
                                color = textSecondary

                                    size = Ux4gButtonSize.Small
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}

private data class SuggestionItem(val title: String, val department: String)
private data class CategoryItem(val label: String, val count: String, val icon: ImageVector)
private data class PopularServiceItem(val title: String, val department: String, val fee: String, val duration: String)

@Composable
private fun CategoryCard(
    item: CategoryItem,
    onClick: () -> Unit,
    cardBg: Color,
    borderCol: Color,
    primaryColor: Color,
    textPrimary: Color,
    textSecondary: Color,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(8.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(8.dp))
            .clickable { onClick() }
            .padding(12.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.Start) {
            Box(
                modifier = Modifier
                    .size(36.dp)
                    .clip(CircleShape)
                    .background(primaryColor.copy(alpha = 0.1f)),
                contentAlignment = Alignment.Center
            ) {
                Icon(
                    imageVector = item.icon,
                    contentDescription = item.label,
                    tint = primaryColor,
                    modifier = Modifier.size(20.dp)
                )
            }
            Spacer(modifier = Modifier.height(8.dp))
            Text(
                text = item.label,
                style = TextStyle(
                    fontSize = 13.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = textPrimary
                )
            )
            Spacer(modifier = Modifier.height(2.dp))
            Text(
                text = item.count,
                style = TextStyle(
                    fontSize = 11.sp,
                    color = textSecondary
                )
            )
        }
    }
}
