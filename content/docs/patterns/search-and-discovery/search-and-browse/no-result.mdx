# No Result

An empathetic and actionable empty state pattern guiding citizens when an active query yields zero direct matches.

Rather than presenting a dead end, the pattern maintains continuity through structured recovery paths: an informative query disclaimer (*"No services found for 'xyz123'"*), phonetic/keyword suggestions (*"Did you mean: Income Certificate, Income & Asset Certificate"*), a direct button to explore the full catalog (*"Browse all services"*), and prominent popular service cards with immediate cost, timeline indicators, and **Apply** triggers.

/**
 * No Result Screen Pattern
 *
 * @param onBack Invoked when citizen taps back button
 * @param onSearchSubmit Invoked when citizen submits a search query
 * @param onSelectSuggestion Invoked when citizen taps a suggested query
 * @param onBrowseAll Invoked when citizen taps 'Browse all services'
 * @param onApplyService Invoked when citizen taps 'Apply' on a recommended service
 */
@Composable
fun NoResultScreen(
    onBack: () -> Unit = {},
    onSearchSubmit: (String) -> Unit = {},
    onSelectSuggestion: (String) -> Unit = {},
    onBrowseAll: () -> Unit = {},
    onApplyService: (String) -> Unit = {}
) {
    val isDark = isSystemInDarkTheme()
    val scrollState = rememberScrollState()

    var searchValue by remember { mutableStateOf("xyz123") }

    val suggestions = listOf(
        "Income Certificate",
        "Income & Asset Certificate"
    )

    val surfaceBg = if (isDark) Ux4gColors.Neutral950 else Ux4gColors.Neutral50
    val cardBg = if (isDark) Ux4gColors.Neutral900 else Color.White
    val borderCol = if (isDark) Ux4gColors.Neutral800 else Color(0xFFE5E7EB)
    val textPrimary = if (isDark) Ux4gColors.Neutral50 else Ux4gColors.Neutral900
    val textSecondary = if (isDark) Ux4gColors.Neutral200 else Ux4gColors.Neutral700
    val primaryPurple = if (isDark) Ux4gColors.Primary300 else Ux4gColors.Primary600
    val greenSuccess = if (isDark) Ux4gColors.Green300 else Ux4gColors.Green600
    val secondaryBtnBg = if (isDark) Ux4gColors.Primary800 else Ux4gColors.Primary100
    val secondaryBtnText = if (isDark) Ux4gColors.Primary200 else Ux4gColors.Primary700

    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(surfaceBg)
    ) {
        // App Header
        Ux4gAppHeader(
            variant = Ux4gAppHeaderVariant.Filled,
            title = "National Services Portal",
            showBackButton = true,
            onBackPressed = onBack
        )

        Spacer(modifier = Modifier.height(16.dp))

        // Search Bar with Submit Button
        Box(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp)
        ) {
            Ux4gSearchField(
                value = searchValue,
                onValueChange = { searchValue = it },
                variant = Ux4gSearchFieldVariant.SearchWithSubmit,
                size = Ux4gSearchFieldSize.Medium,
                placeholder = "Search services...",
                showVoiceIcon = true,
                showClearIcon = true,
                buttonStyle = Ux4gSearchFieldButtonStyle.Filled,
                onSubmitClick = { onSearchSubmit(searchValue) }
            )
        }

        // Empty State & Fallback Recommendations
        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(scrollState)
                .padding(16.dp),
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            Spacer(modifier = Modifier.height(28.dp))

            // Primary Search Icon
            Icon(
                imageVector = Icons.Default.Search,
                contentDescription = "No Results Found",
                modifier = Modifier.size(52.dp),
                tint = primaryPurple
            )

            Spacer(modifier = Modifier.height(16.dp))

            // Empty State Heading
            Text(
                text = "No services found for \\"" + searchValue + "\\"",
                style = TextStyle(
                    fontSize = 16.sp,
                    fontWeight = FontWeight.Bold,
                    color = textPrimary,
                    textAlign = TextAlign.Center
                )
            )

            Spacer(modifier = Modifier.height(12.dp))

            // Did you mean prompt
            Text(
                text = "Did you mean:",
                style = TextStyle(
                    fontSize = 13.sp,
                    color = textSecondary
                )
            )

            Spacer(modifier = Modifier.height(6.dp))

            // Suggestions List
            Row(
                horizontalArrangement = Arrangement.Center,
                verticalAlignment = Alignment.CenterVertically
            ) {
                suggestions.forEachIndexed { index, suggestion ->
                    val isLast = index == suggestions.size - 1
                    Text(
                        text = if (isLast) suggestion else suggestion + ",  ",
                        style = TextStyle(
                            fontSize = 13.sp,
                            fontWeight = FontWeight.Medium,
                            color = primaryPurple
                        ),
                        modifier = Modifier.clickable {
                            searchValue = suggestion
                            onSelectSuggestion(suggestion)
                        }
                    )
                }
            }

            Spacer(modifier = Modifier.height(22.dp))

            // Browse All Services Button
            Ux4gButton(
                text = "Browse all services",
                onClick = onBrowseAll,
                variant = Ux4gButtonVariant.Secondary,
                size = Ux4gButtonSize.Medium,
                backgroundColor = secondaryBtnBg,
                contentColor = secondaryBtnText
            )

            Spacer(modifier = Modifier.height(44.dp))

            // Popular Services Section Header
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.Start
            ) {
                Text(
                    text = "Popular services",
                    style = TextStyle(
                        fontSize = 16.sp,
                        fontWeight = FontWeight.Bold,
                        color = textPrimary
                    )
                )
            }

            Spacer(modifier = Modifier.height(12.dp))

            // Popular Service Card 1: Income Certificate
            PopularServiceCard(
                title = "Income Certificate",
                department = "Revenue Department",
                fee = "Free",
                timeline = "20 mins",
                cardBg = cardBg,
                borderCol = borderCol,
                textPrimary = textPrimary,
                textSecondary = textSecondary,
                greenSuccess = greenSuccess,
                onApply = { onApplyService("Income Certificate") }
            )

            Spacer(modifier = Modifier.height(12.dp))

            // Popular Service Card 2: Ration Card
            PopularServiceCard(
                title = "Ration Card",
                department = "Food & Civil Supplies",
                fee = "Free",
                timeline = "10 days",
                cardBg = cardBg,
                borderCol = borderCol,
                textPrimary = textPrimary,
                textSecondary = textSecondary,
                greenSuccess = greenSuccess,
                onApply = { onApplyService("Ration Card") }
            )

            Spacer(modifier = Modifier.height(24.dp))
        }
    }
}

/**
 * Reusable Popular Service Recommendation Card
 */
@Composable
private fun PopularServiceCard(
    title: String,
    department: String,
    fee: String,
    timeline: String,
    cardBg: Color,
    borderCol: Color,
    textPrimary: Color,
    textSecondary: Color,
    greenSuccess: Color,
    onApply: () -> Unit
) {
    Box(
        modifier = Modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(cardBg)
            .border(1.dp, borderCol, RoundedCornerShape(12.dp))
            .padding(14.dp)
    ) {
        Column(modifier = Modifier.fillMaxWidth()) {
            Text(
                text = title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = textPrimary
                )
            )

            Spacer(modifier = Modifier.height(2.dp))

            Text(
                text = department,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = textSecondary
                )
            )

            Spacer(modifier = Modifier.height(10.dp))

            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = fee,
                    style = TextStyle(
                        fontSize = 12.sp,
                        fontWeight = FontWeight.SemiBold,
                        color = greenSuccess
                    )
                )

                Text(
                    text = "  ·  " + timeline,
                    style = TextStyle(
                        fontSize = 12.sp,
                        color = textSecondary
                    )
                )
            }

            Spacer(modifier = Modifier.height(12.dp))

            Ux4gButton(
                text = "Apply",
                onClick = onApply,
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Small
            )
        }
    }
}
