# Service Detail

A comprehensive service detail pattern delivering complete statutory transparency, eligibility criteria, required documentation, and direct application pathways for individual government services and citizen schemes.

The screen establishes contextual continuity through an intuitive breadcrumb path (`Home › Certificates › Birth Certificate`). A prominent hero service card communicates essential parameters at a glance: issuing authority (*Municipal Corporation*), fee status (*Free*), standard processing turnaround (*20 mins online*), and a trust-reinforcing **Government verified** indicator, paired with a full-width primary **Apply now** trigger (`Ux4gButton`). Segmented tab navigation (*Overview*, *Eligibility*, *Documents*, *Process*) organizes extensive statutory documentation into digestible sections. Inside the overview panel, structured information callout boxes highlight key conditions, while cross-linked related services foster holistic citizen discovery.

 * @param onApplyRelated Invoked when citizen applies for a related service

fun ServiceDetailScreen(
    onBack: () -> Unit = {},
    onApply: () -> Unit = {},
    onTabSelect: (Int) -> Unit = {},
    onApplyRelated: (String) -> Unit = {}

                title = "Required documents",

            ServiceInfoSection(
                title = "Fee",
                description = "Free within 21 days of birth. ₹20 late fee applies thereafter."
            ),
            ServiceInfoSection(
                title = "Processing time",
                description = "20 minutes online · up to 7 working days if applied offline."
            )
        )
    }

        listOf(
            RelatedService("Death Certificate", "Municipal Corporation", "Free", "20 mins", isFree = true),
            RelatedService("Marriage Certificate", "Registrar of Marriages", "₹100", "15 days", isFree = false),
            RelatedService("Aadhaar Enrolment", "UIDAI", "Free", "90 days", isFree = true)
        )

        modifier = Modifier
            .fillMaxSize()
            .background(colors.screenBackground)

        // App Header - filled purple
        Ux4gAppHeader(
            title = "National Services Portal",
            variant = Ux4gAppHeaderVariant.Filled,
            showBackButton = true,
            onBackPressed = onBack,
            backgroundColor = Ux4gColors.primary600,
            modifier = Modifier.fillMaxWidth()
        )

        // Scrollable Content
        Column(
            modifier = Modifier
                .fillMaxSize()
                .verticalScroll(scrollState)
        ) {
            // Breadcrumbs Navigation
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier
                    .fillMaxWidth()
                    .background(colors.surfaceBackground)
                    .padding(horizontal = 16.dp, vertical = 10.dp)
            ) {
                Icon(

                    contentDescription = "Home",
                    tint = colors.subtleText,

                Spacer(modifier = Modifier.width(4.dp))
                Text(
                    text = "Home",
                    style = TextStyle(fontSize = 12.sp, color = colors.subtleText)

                    style = TextStyle(fontSize = 12.sp, color = colors.subtleText)

                    text = "Birth Certificate",

                        fontWeight = FontWeight.SemiBold,
                        color = colors.breadcrumbActive

                    .fillMaxWidth()

                    .clip(RoundedCornerShape(12.dp))

                    text = "Related services",

                        fontSize = 16.sp,

}
