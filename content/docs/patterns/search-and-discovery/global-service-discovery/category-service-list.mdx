# Category Service List

A sector-specific service directory pattern designed for focused exploration of public schemes, welfare programs, and statutory certifications belonging to a selected domain such as *Health*, *Agriculture*, *Education*, or *Social Welfare*.

The screen establishes orientation with an intuitive breadcrumb path (`Home › Services › Health`), followed by an informative heading outlining the catalog scope. Citizens can evaluate aggregate counts (*"48 services"*), trigger granular modal filters (`Ux4gButton` with `Icons.Default.Tune`), and horizontally pivot across specialized service clusters via filter chips. Each service card clearly highlights fee structures (*Free* or *Paid* tags), statutory turnaround SLAs, and a direct **Apply** call to action, accompanied by an accordion disclosure drawer revealing granular eligibility benchmarks, departmental contacts, and requisite documentation.

@Composable
private fun ExpandableServiceCard(
    service: ServiceListItem,
    isExpanded: Boolean,
    onToggleExpand: () -> Unit,
    colors: CategoryServiceListColors,
    onApply: () -> Unit,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(10.dp))
            .background(colors.surfaceBackground)
            .border(1.dp, colors.border, RoundedCornerShape(10.dp))
            .padding(14.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.start) {
            Row(
                verticalAlignment = Alignment.CenterVertically,
                modifier = Modifier.fillMaxWidth()
            ) {
                Column(modifier = Modifier.weight(1f)) {
                    Text(
                        text = service.title,
                        style = TextStyle(
                            fontSize = 14.sp,
                            fontWeight = FontWeight.Bold,
                            color = colors.title
                        )
                    )

                    Spacer(modifier = Modifier.height(6.dp))

                    Row(verticalAlignment = Alignment.CenterVertically) {
                        Box(
                            modifier = Modifier
                                .clip(RoundedCornerShape(4.dp))
                                .background(colors.tagBg)
                                .padding(horizontal = 6.dp, vertical = 2.dp)
                        ) {
                            Text(
                                text = service.badge,
                                style = TextStyle(
                                    fontSize = 11.sp,
                                    fontWeight = FontWeight.Medium,
                                    color = colors.tagText
                                )
                            )
                        }

                        if (service.extraFee != null) {
                            Spacer(modifier = Modifier.width(6.dp))
                            Text(
                                text = service.extraFee,
                                style = TextStyle(
                                    fontSize = 11.sp,
                                    color = colors.subtleText
                                )
                            )
                        }

                        Spacer(modifier = Modifier.width(8.dp))

                        Icon(
                            imageVector = Icons.Default.AccessTime,
                            contentDescription = "Turnaround Time",
                            tint = colors.timeIcon,
                            modifier = Modifier.size(13.dp)
                        )

                        Spacer(modifier = Modifier.width(3.dp))

                        Text(
                            text = service.time,
                            style = TextStyle(
                                fontSize = 11.sp,
                                color = colors.subtleText
                            )
                        )
                    }
                }

                Spacer(modifier = Modifier.width(8.dp))

                Ux4gButton(
                    text = "Apply",
                    onClick = onApply,
                    variant = Ux4gButtonVariant.Outline,
                    size = Ux4gButtonSize.Small
                )

                Spacer(modifier = Modifier.width(4.dp))

                Box(
                    modifier = Modifier
                        .clickable { onToggleExpand() }
                        .padding(4.dp)
                ) {
                    Icon(
                        imageVector = if (isExpanded) Icons.Default.KeyboardArrowUp else Icons.Default.KeyboardArrowDown,
                        contentDescription = if (isExpanded) "Collapse" else "Expand",
                        tint = colors.subtleText,
                        modifier = Modifier.size(20.dp)
                    )
                }
            }

            // Accordion Drawer Content
            AnimatedVisibility(
                visible = isExpanded && service.details != null,
                enter = fadeIn() + expandVertically(),
                exit = fadeOut() + shrinkVertically()
            ) {
                Column {
                    Spacer(modifier = Modifier.height(12.dp))
                    Divider(color = colors.dividerColor, thickness = 1.dp)
                    Spacer(modifier = Modifier.height(12.dp))

                    service.details?.forEach { row ->
                        Row(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(bottom = 8.dp)
                        ) {
                            Column(modifier = Modifier.weight(1f)) {
                                Text(
                                    text = row.first.first,
                                    style = TextStyle(
                                        fontSize = 11.sp,
                                        color = colors.mutedText
                                    )
                                )
                                Spacer(modifier = Modifier.height(2.dp))
                                Text(
                                    text = row.first.second,
                                    style = TextStyle(
                                        fontSize = 12.sp,
                                        fontWeight = FontWeight.Medium,
                                        color = colors.title
                                    )
                                )
                            }

                            Spacer(modifier = Modifier.width(12.dp))

                            Column(modifier = Modifier.weight(1f)) {
                                Text(
                                    text = row.second.first,
                                    style = TextStyle(
                                        fontSize = 11.sp,
                                        color = colors.mutedText
                                    )
                                )
                                Spacer(modifier = Modifier.height(2.dp))
                                Text(
                                    text = row.second.second,
                                    style = TextStyle(
                                        fontSize = 12.sp,
                                        fontWeight = FontWeight.Medium,
                                        color = colors.title
                                    )
                                )
                            }
                        }
                    }
                }
            }
        }
    }
}
