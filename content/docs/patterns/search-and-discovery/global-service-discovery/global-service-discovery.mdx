# Global Service Discovery

A unified public service portal landing pattern designed to accelerate citizen service discovery, entitlements navigation, and public welfare program access across national and state government catalogs.

The screen anchors discovery with an omnichannel search bar (`Ux4gSearchField` with keyword input, submit trigger, voice dictation, and autocomplete readiness), followed by a 2-column thematic category grid (*Health*, *Education*, *Agriculture*, *Land Records*, *Transport*, *Social Welfare*, *Utilities*, *Finance*). Prominent discovery sections highlight timely national interventions (*Featured Scheme* with direct eligibility assessment), high-volume public services (*Popular Services* with fee, statutory turnaround time, and direct apply actions), targeted citizen benefits (*Recommended for User*), and an intuitive assessment callout (*Find my schemes* quiz) to assist citizens uncertain of their entitlements.

@Composable
private fun CategoryTile(
    category: DiscoveryCategory,
    colors: GlobalServiceDiscoveryColors,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .clip(RoundedCornerShape(8.dp))
            .background(colors.surfaceBackground)
            .border(1.dp, colors.border, RoundedCornerShape(8.dp))
            .clickable { onClick() }
            .padding(12.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.start) {
            Box(
                modifier = Modifier
                    .size(36.dp)
                    .clip(CircleShape)
                    .background(colors.categoryIconBg),
                contentAlignment = Alignment.Center
            ) {
                Icon(
                    imageVector = category.icon,
                    contentDescription = category.label,
                    tint = colors.categoryIconTint,
                    modifier = Modifier.size(20.dp)
                )
            }

            Spacer(modifier = Modifier.height(10.dp))

            Text(
                text = category.label,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
                style = TextStyle(
                    fontSize = 13.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = colors.title
                )
            )

            Spacer(modifier = Modifier.height(2.dp))

            Text(
                text = category.count,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis,
                style = TextStyle(
                    fontSize = 11.sp,
                    color = colors.subtleText
                )
            )
        }
    }
}

@Composable
private fun ServiceItemCard(
    title: String,
    department: String,
    fee: String,
    metadata: String,
    isFree: Boolean,
    colors: GlobalServiceDiscoveryColors,
    onApply: () -> Unit,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .fillMaxWidth()
            .clip(RoundedCornerShape(12.dp))
            .background(colors.surfaceBackground)
            .border(1.dp, colors.border, RoundedCornerShape(12.dp))
            .padding(14.dp)
    ) {
        Column(crossAxisAlignment = CrossAxisAlignment.start) {
            Text(
                text = title,
                style = TextStyle(
                    fontSize = 14.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.title
                )
            )

            Spacer(modifier = Modifier.height(2.dp))

            Text(
                text = department,
                style = TextStyle(
                    fontSize = 12.sp,
                    color = colors.subtleText
                )
            )

            Spacer(modifier = Modifier.height(8.dp))

            Row(verticalAlignment = Alignment.CenterVertically) {
                Text(
                    text = fee,
                    style = TextStyle(
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Medium,
                        color = if (isFree) colors.freeBadgeColor else colors.title
                    )
                )
                Text(
                    text = "  ·  $metadata",
                    style = TextStyle(
                        fontSize = 12.sp,
                        color = colors.subtleText
                    )
                )
            }

            Spacer(modifier = Modifier.height(10.dp))

            Ux4gButton(
                text = "Apply",
                onClick = onApply,
                variant = Ux4gButtonVariant.Outline,
                size = Ux4gButtonSize.Small
            )
        }
    }
}
