# Manage All Subscriptions

A centralized notification management pattern providing citizens with a comprehensive overview of every alert type sent by the portal. To maintain strict regulatory compliance while honoring citizen autonomy, alerts are divided into:
- **Optional Alerts**: Customizable subscriptions (*Status updates*, *Action reminders*, *Certificate expiry*, *Promotional updates*) with individual `Ux4gSwitch` toggles and an immediate **Turn off all optional notifications** bulk action.
- **Mandatory Policy Alerts**: Statutory communications (*SLA breach alerts*, *Rejection notices*) required under government service guarantee frameworks, surfaced with non-removable **Always on** badges.

@Composable
private fun ToggleItemRow(
    title: String,
    subtitle: String,
    checked: Boolean,
    onCheckedChange: (Boolean) -> Unit,
    colors: ManageAllSubscriptionsColors
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Column(
            modifier = Modifier
                .weight(1f)
                .padding(end = 16.dp)
        ) {
            Text(
                text = title,
                style = TextStyle(
                    fontSize = 15.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = colors.title
                )
            )
            Spacer(Modifier.height(2.dp))
            Text(
                text = subtitle,
                style = TextStyle(
                    fontSize = 13.sp,
                    color = colors.subtleText,
                    lineHeight = 18.sp
                )
            )
        }

        Ux4gSwitch(
            checked = checked,
            onCheckedChange = onCheckedChange,
            size = Ux4gSwitchSize.M,
            labelPosition = Ux4gSwitchLabelPosition.NoLabel
        )
    }
}
