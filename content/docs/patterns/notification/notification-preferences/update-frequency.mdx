# Update Frequency

A notification delivery frequency preference pattern allowing citizens to control how often alerts and updates are batched and delivered. The pattern offers mutually exclusive cadence options—**Immediately** (instant real-time dispatch), **Daily Summary** (one consolidated digest at 6:00 PM), and **Weekly Digest** (comprehensive roundup every Monday morning)—contained within an elevated card using selectable `Ux4gRadioButton` cards, filter tabs, and auto-saving state.

@Composable
fun UpdateFrequencyScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    onMenuClick: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkUpdateFrequencyColors else LightUpdateFrequencyColors

    var selectedTab by rememberSaveable { mutableStateOf("Update Frequency") }
    var selectedCadence by rememberSaveable { mutableStateOf(NotificationFrequencyCadence.Immediately) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 20.dp)
        ) {
            // Screen Title
            Text(
                text = "Notification Preferences",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            // Subtitle Description
            Text(
                text = "Control how and when the portal contacts you. Changes are saved automatically.",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 20.sp
                )
            )
            Spacer(Modifier.height(18.dp))

            // Filter Tabs Bar
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .horizontalScroll(rememberScrollState()),
                horizontalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                listOf("Update Frequency", "Per Service Preferences", "Quiet Hours").forEach { tab ->
                    val isSelected = selectedTab == tab
                    Box(
                        modifier = Modifier
                            .clip(RoundedCornerShape(8.dp))
                            .background(
                                if (isSelected) colors.activeTabBackground
                                else colors.inactiveTabBackground
                            )
                            .border(
                                width = 1.dp,
                                color = if (isSelected) colors.activeTabBackground else colors.inactiveTabBorder,
                                shape = RoundedCornerShape(8.dp)
                            )
                            .clickable { selectedTab = tab }
                            .padding(horizontal = 16.dp, vertical = 10.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = tab,
                            style = TextStyle(
                                fontSize = 14.sp,
                                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Medium,
                                color = if (isSelected) colors.activeTabText else colors.inactiveTabText
                            )
                        )
                    }
                }
            }
            Spacer(Modifier.height(20.dp))

            // Elevated Frequency Card
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .clip(RoundedCornerShape(16.dp))
                    .background(colors.cardBackground)
                    .border(1.dp, colors.border, RoundedCornerShape(16.dp))
                    .padding(20.dp)
            ) {
                Column(modifier = Modifier.fillMaxWidth()) {
                    Text(
                        text = "Update frequency",
                        style = TextStyle(
                            fontSize = 20.sp,
                            fontWeight = FontWeight.Bold,
                            color = colors.title
                        )
                    )
                    Spacer(Modifier.height(6.dp))
                    Text(
                        text = "Decide how often the portal batches and sends your notifications.",
                        style = TextStyle(
                            fontSize = 14.sp,
                            color = colors.subtleText,
                            lineHeight = 20.sp
                        )
                    )
                    Spacer(Modifier.height(20.dp))

                    // Cadence Option Cards
                    Column(verticalArrangement = Arrangement.spacedBy(14.dp)) {
                        NotificationFrequencyCadence.values().forEach { cadence ->
                            val isSelected = selectedCadence == cadence

                            Box(
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .clip(RoundedCornerShape(12.dp))
                                    .background(
                                        if (isSelected) colors.selectedOptionBackground
                                        else colors.unselectedOptionBackground
                                    )
                                    .border(
                                        width = if (isSelected) 1.5.dp else 1.dp,
                                        color = if (isSelected) colors.selectedOptionBorder
                                        else colors.unselectedOptionBorder,
                                        shape = RoundedCornerShape(12.dp)
                                    )
                                    .clickable { selectedCadence = cadence }
                                    .padding(16.dp)
                            ) {
                                Row(
                                    verticalAlignment = Alignment.CenterVertically,
                                    modifier = Modifier.fillMaxWidth()
                                ) {
                                    Ux4gRadioButton(
                                        selected = isSelected,
                                        onClick = { selectedCadence = cadence },
                                        size = Ux4gRadioButtonSize.Medium
                                    )
                                    Spacer(Modifier.width(14.dp))
                                    Column(modifier = Modifier.weight(1f)) {
                                        Text(
                                            text = cadence.title,
                                            style = TextStyle(
                                                fontSize = 16.sp,
                                                fontWeight = FontWeight.SemiBold,
                                                color = if (isSelected) colors.selectedOptionTitle else colors.title
                                            )
                                        )
                                        Spacer(Modifier.height(4.dp))
                                        Text(
                                            text = cadence.description,
                                            style = TextStyle(
                                                fontSize = 14.sp,
                                                color = if (isSelected) colors.selectedOptionSubtitle else colors.subtleText,
                                                lineHeight = 19.sp
                                            )
                                        )
                                    }
                                }
                            }
                        }
                    }
                }
            }
            Spacer(Modifier.height(20.dp))
        }

        BrandFooter(colors = colors)
    }
}

@Composable
private fun BrandHeader(colors: UpdateFrequencyColors, onMenuClick: () -> Unit) {
    Column {
        Ux4gAppHeader(
            title = "",
            variant = Ux4gAppHeaderVariant.Light,
            height = 64.dp,
            horizontalPadding = Ux4gSpace.Space16,
            leadingSpacing = Ux4gSpace.Space8,
            leadingWidgets = listOf(
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_national_emblem),
                        contentDescription = "National Emblem of India",
                        colorFilter = colors.emblemTint?.let { ColorFilter.tint(it) },
                        modifier = Modifier.height(32.dp)
                    )
                },
                {
                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp
                    )
                },
                {
                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),
                        contentDescription = "Government of India",
                        colorFilter = ColorFilter.tint(colors.brandLogoTint),
                        modifier = Modifier.height(32.dp)
                    )
                }
            ),
            actions = listOf(
                Ux4gAppHeaderAction(
                    customWidget = {
                        Box(
                            modifier = Modifier
                                .size(40.dp)
                                .clip(RoundedCornerShape(10.dp))
                                .background(colors.menuButtonBackground)
                                .border(1.5.dp, colors.menuButtonBorder, RoundedCornerShape(10.dp))
                                .clickable(onClick = onMenuClick),
                            contentAlignment = Alignment.Center
                        ) {
                            Icon(
                                imageVector = Icons.Filled.Menu,
                                contentDescription = "Open menu",
                                tint = colors.menuButtonIcon,
                                modifier = Modifier.size(20.dp)
                            )
                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}

@Composable
private fun BrandFooter(colors: UpdateFrequencyColors) {
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .padding(top = Ux4gSpace.Space8, bottom = Ux4gSpace.Space20),
        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = "Powered by -",
            style = Ux4gTheme.typography.lS_default.copy(color = colors.mutedText)
        )
        Spacer(Modifier.width(Ux4gSpace.Space8))
        Image(
            painter = painterResource(AppR.drawable.ic_digital_india),
            contentDescription = "Digital India",
            colorFilter = colors.footerLogoTint?.let { ColorFilter.tint(it) },
            modifier = Modifier.height(22.dp)
        )
    }
}
