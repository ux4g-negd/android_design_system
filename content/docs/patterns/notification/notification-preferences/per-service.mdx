# Per Service Preferences

A granular notification customization pattern allowing citizens to configure delivery preferences on a per-service basis. Rather than global channel settings, citizens can fine-tune alert subscriptions for specific enrolled schemes and applications—such as **Income Certificate**, **PAN Card Correction**, **Driving License Renewal**, and **Birth Certificate Registration**—controlling which services trigger active updates and milestone tracking.

@Composable
fun PerServicePreferencesScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    onMenuClick: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkPerServicePreferencesColors else LightPerServicePreferencesColors

    var selectedTab by rememberSaveable { mutableStateOf("Per Service Preferences") }
    var incomeCertificateAlerts by rememberSaveable { mutableStateOf(true) }
    var panCorrectionAlerts by rememberSaveable { mutableStateOf(true) }
    var drivingLicenseAlerts by rememberSaveable { mutableStateOf(true) }
    var birthCertificateAlerts by rememberSaveable { mutableStateOf(false) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 20.dp)
        ) {
            // Screen Title
            Text(
                text = "Notification Preferences",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            // Subtitle Description
            Text(
                text = "Control how and when the portal contacts you. Changes are saved automatically.",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 20.sp
                )
            )
            Spacer(Modifier.height(18.dp))

            // Filter Tabs Bar
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .horizontalScroll(rememberScrollState()),
                horizontalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                listOf("Notification Channels", "Update Frequency", "Per Service Preferences").forEach { tab ->
                    val isSelected = selectedTab == tab
                    Box(
                        modifier = Modifier
                            .clip(RoundedCornerShape(8.dp))
                            .background(
                                if (isSelected) colors.activeTabBackground
                                else colors.inactiveTabBackground
                            )
                            .border(
                                width = 1.dp,
                                color = if (isSelected) colors.activeTabBackground else colors.inactiveTabBorder,
                                shape = RoundedCornerShape(8.dp)
                            )
                            .clickable { selectedTab = tab }
                            .padding(horizontal = 16.dp, vertical = 10.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = tab,
                            style = TextStyle(
                                fontSize = 14.sp,
                                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Medium,
                                color = if (isSelected) colors.activeTabText else colors.inactiveTabText
                            )
                        )
                    }
                }
            }

                    .background(colors.cardBackground)
                    .border(1.dp, colors.border, RoundedCornerShape(16.dp))
                    .padding(20.dp)
            ) {

@Composable
private fun ServiceToggleRow(
    title: String,
    subtitle: String,
    checked: Boolean,
    onCheckedChange: (Boolean) -> Unit,
    colors: PerServicePreferencesColors
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Column(modifier = Modifier.weight(1f).padding(end = 16.dp)) {
            Text(
                text = title,
                style = TextStyle(
                    fontSize = 16.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = colors.title
                )
            )
            Spacer(Modifier.height(4.dp))
            Text(
                text = subtitle,
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 19.sp
                )
            )
        }

        Ux4gSwitch(
            checked = checked,
            onCheckedChange = onCheckedChange,
            size = Ux4gSwitchSize.M,
            labelPosition = Ux4gSwitchLabelPosition.NoLabel
        )
    }
}

                    Ux4gDivider(
                        orientation = Ux4gDividerOrientation.Vertical,
                        color = colors.headerDivider,
                        height = 28.dp

                    Image(
                        painter = painterResource(Ux4gR.drawable.ic_union),

                            modifier = Modifier

                        }
                    }
                )
            )
        )
        Ux4gDivider(color = colors.border)
    }
}

        horizontalArrangement = Arrangement.Center,
        verticalAlignment = Alignment.CenterVertically
    ) {
        Text(
            text = "Powered by -",
            style = Ux4gTheme.typography.lS_default.copy(color = colors.mutedText)
        )
        Spacer(Modifier.width(Ux4gSpace.Space8))
        Image(
            painter = painterResource(AppR.drawable.ic_digital_india),
            contentDescription = "Digital India",
            colorFilter = colors.footerLogoTint?.let { ColorFilter.tint(it) },
            modifier = Modifier.height(22.dp)
        )
    }
}
