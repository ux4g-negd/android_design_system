# Notification Channels

A central notification configuration pattern allowing citizens to customize multi-channel delivery preferences (SMS, Email, in-app push alerts, and WhatsApp). The screen provides instant binary toggle controls for each communication channel, verified destination metadata (registered phone number and masked email address), regulatory opt-in notices, category filter navigation (*"Notification Channels"*, *"Update Frequency"*), and an official government brand header and footer.

@Composable
fun NotificationChannelsScreen(
    modifier: Modifier = Modifier,
    isDarkTheme: Boolean = isSystemInDarkTheme(),
    onMenuClick: () -> Unit = {}
) {
    val colors = if (isDarkTheme) DarkNotificationChannelsColors else LightNotificationChannelsColors

    var selectedTab by rememberSaveable { mutableStateOf("Notification Channels") }
    var smsEnabled by rememberSaveable { mutableStateOf(true) }
    var emailEnabled by rememberSaveable { mutableStateOf(true) }
    var appPushEnabled by rememberSaveable { mutableStateOf(true) }
    var whatsappEnabled by rememberSaveable { mutableStateOf(true) }

    Column(
        modifier = modifier
            .fillMaxSize()
            .background(colors.screenBackground)
    ) {
        BrandHeader(colors = colors, onMenuClick = onMenuClick)

        Column(
            modifier = Modifier
                .weight(1f)
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 16.dp, vertical = 20.dp)
        ) {
            // Screen Title
            Text(
                text = "Notification Preferences",
                style = TextStyle(
                    fontSize = 24.sp,
                    fontWeight = FontWeight.Bold,
                    color = colors.title,
                    letterSpacing = (-0.3).sp
                )
            )
            Spacer(Modifier.height(8.dp))

            // Subtitle Description
            Text(
                text = "Control how and when the portal contacts you. Changes are saved automatically.",
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 20.sp
                )
            )
            Spacer(Modifier.height(18.dp))

            // Filter Tabs Bar
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .horizontalScroll(rememberScrollState()),
                horizontalArrangement = Arrangement.spacedBy(10.dp)
            ) {
                listOf("Notification Channels", "Update Frequency", "Quiet Hours").forEach { tab ->
                    val isSelected = selectedTab == tab
                    Box(
                        modifier = Modifier
                            .clip(RoundedCornerShape(8.dp))
                            .background(
                                if (isSelected) colors.activeTabBackground
                                else colors.inactiveTabBackground
                            )
                            .border(
                                width = 1.dp,
                                color = if (isSelected) colors.activeTabBackground else colors.inactiveTabBorder,
                                shape = RoundedCornerShape(8.dp)
                            )
                            .clickable { selectedTab = tab }
                            .padding(horizontal = 16.dp, vertical = 10.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = tab,
                            style = TextStyle(
                                fontSize = 14.sp,
                                fontWeight = if (isSelected) FontWeight.SemiBold else FontWeight.Medium,
                                color = if (isSelected) colors.activeTabText else colors.inactiveTabText
                            )

                    .background(colors.cardBackground)

@Composable
private fun ChannelToggleRow(
    title: String,
    subtitle: String,
    checked: Boolean,
    onCheckedChange: (Boolean) -> Unit,
    colors: NotificationChannelsColors
) {
    Row(
        modifier = Modifier.fillMaxWidth(),
        verticalAlignment = Alignment.CenterVertically,
        horizontalArrangement = Arrangement.SpaceBetween
    ) {
        Column(modifier = Modifier.weight(1f).padding(end = 16.dp)) {
            Text(
                text = title,
                style = TextStyle(
                    fontSize = 16.sp,
                    fontWeight = FontWeight.SemiBold,
                    color = colors.title
                )
            )
            Spacer(Modifier.height(4.dp))
            Text(
                text = subtitle,
                style = TextStyle(
                    fontSize = 14.sp,
                    color = colors.subtleText,
                    lineHeight = 19.sp
                )
            )
        }

        Ux4gSwitch(
            checked = checked,
            onCheckedChange = onCheckedChange,
            size = Ux4gSwitchSize.M,
            labelPosition = Ux4gSwitchLabelPosition.NoLabel
        )
    }
}

        )
    }
}
